## 1. 概述

### 1.1 Purpose

This framework ensures code and documentation remain synchronized throughout the project lifecycle. It provides:
- Automated documentation generation from code
- Code validation against documentation
- Bidirectional traceability
- Consistency enforcement

### 1.2 Principles

| Principle | 描述 |
|-----------|-------------|
| Single Source of Truth | Documentation and code derive from shared specifications |
| Automated Sync | Reduce manual effort through automation |
| Fail-Fast | Catch inconsistencies early in the pipeline |
| Bidirectional Traceability | Trace requirements → code → tests → documentation |

## 4. Traceability Matrix

### 4.1 需求可追溯性

**PRD → Code → Tests Mapping:**

```python
# traceability.py
class TraceabilityMatrix:
    def __init__(self):
        self.requirements = {}
        self.implementations = {}
        self.tests = {}
    
    def add_requirement(self, req_id, description, source):
        """Add PRD requirement."""
        self.requirements[req_id] = {
            'description': description,
            'source': source,
            'implementations': [],
            'tests': []
        }
    
    def link_implementation(self, req_id, file_path, line_number):
        """Link requirement to code implementation."""
        if req_id in self.requirements:
            self.requirements[req_id]['implementations'].append({
                'file': file_path,
                'line': line_number
            })
    
    def link_test(self, req_id, test_id, test_file):
        """Link requirement to test case."""
        if req_id in self.requirements:
            self.requirements[req_id]['tests'].append({
                'test_id': test_id,
                'file': test_file
            })
    
    def get_coverage_report(self):
        """Generate coverage report."""
        covered = 0
        total = len(self.requirements)
        
        for req_id, req in self.requirements.items():
            if req['implementations'] and req['tests']:
                covered += 1
        
        return {
            'total': total,
            'covered': covered,
            'coverage_percent': (covered / total) * 100 if total > 0 else 0
        }
```

## 6. 版本对齐

### 6.1 文档版本跟踪

```python
# version_tracker.py
class DocumentVersionTracker:
    def __init__(self):
        self.versions = {}
    
    def register_document(self, doc_id, version, dependencies=None):
        """Register document with version and dependencies."""
        self.versions[doc_id] = {
            'version': version,
            'dependencies': dependencies or [],
            'last_updated': datetime.now()
        }
    
    def check_compatibility(self, doc_id, required_versions):
        """Check if document versions are compatible."""
        for dep_id, required_version in required_versions.items():
            if dep_id not in self.versions:
                return False, f"Missing dependency: {dep_id}"
            
            current = self.versions[dep_id]['version']
            if not is_compatible(current, required_version):
                return False, f"Version mismatch: {dep_id} requires {required_version}, has {current}"
        
        return True, "All versions compatible"
```

### 6.2 版本兼容性矩阵

| Document | 版本 | Compatible With |
|----------|---------|-----------------|
| PRD | v0.5.1 | DB Design v1.0, Calc Spec v1.1 |
| DB Design | v1.0 | PRD v0.5.1, API Design v1.0 |
| Calc Spec | v1.1 | PRD v0.5.1, DB Design v1.0 |
| API Design | v1.0 | DB Design v1.0, UI Design v1.0 |

### 6.3 版本更新工作流

```
1. Update source document (e.g., PRD)
   ↓
2. Run compatibility check
   ↓
3. If compatible: Update dependent documents
   If incompatible: Create migration plan
   ↓
4. Update version numbers
   ↓
5. Update cross-references
   ↓
6. Run validation
   ↓
7. Commit changes
```

import re

def validate_links():
    """Check all markdown links are valid."""
    errors = []
    
    for md_file in Path('.').rglob('*.md'):
        content = md_file.read_text()
        
        # Find all markdown links
        links = re.findall(r'\[([^\]]+)\]\(([^)]+)\)', content)

### 8.2 需求覆盖率检查器

```python
# scripts/check_requirement_coverage.py
"""Check all PRD requirements are covered by tests."""

import re
from pathlib import Path

def extract_prd_requirements():
    """Extract all requirement IDs from PRD."""
    requirements = set()
    
    prd_file = Path('PCB工厂冷源仿真平台_PRD_v0.5.1_修订版.md')
    content = prd_file.read_text()
    
    # Find requirement IDs (e.g., F1-001, F2-015)
    req_pattern = r'F\d+-\d+'
    requirements = set(re.findall(req_pattern, content))
    
    return requirements

def extract_test_requirements():
    """Extract requirement IDs covered by tests."""
    covered = set()
    
    for test_file in Path('tests').rglob('*.py'):
        content = test_file.read_text()
        
        # Find requirement references
        req_pattern = r'(?:PRD|Requirement)[:\s]+(F\d+-\d+)'
        matches = re.findall(req_pattern, content)
        covered.update(matches)
    
    return covered

def check_coverage():
    """Check requirement coverage."""
    prd_reqs = extract_prd_requirements()
    test_reqs = extract_test_requirements()
    
    uncovered = prd_reqs - test_reqs
    
    if uncovered:
        print(f"Uncovered requirements: {len(uncovered)}")
        for req in sorted(uncovered):
            print(f"  - {req}")
        return False
    
    print(f"All {len(prd_reqs)} requirements covered by tests")
    return True

if __name__ == '__main__':
    if not check_coverage():
        exit(1)
```
