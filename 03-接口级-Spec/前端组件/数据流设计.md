```
用户输入用户名密码
      │
      ▼
POST /api/v1/auth/login/
      │
      ├── 成功
      │   ├── 保存 Access Token 到内存（Context）
      │   ├── 保存 Refresh Token 到 localStorage
      │   ├── 获取用户信息
      │   └── 跳转到地图首页
      │
      └── 失败（401）
          └── 显示错误提示"账号或密码错误"
```

**代码示例：**
```typescript
// src/pages/Login/index.tsx
const LoginPage = () => {
  const { login } = useAuth();
  const navigate = useNavigate();
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (values: { username: string; password: string }) => {
    setLoading(true);
    try {
      await login(values.username, values.password);
      message.success('登录成功');
      navigate('/map');

      setLoading(false);
    }

      <Form.Item name="username" rules={[{ required: true }]}>
        <Input placeholder="用户名" />
      </Form.Item>

        <Input.Password placeholder="密码" />

  );
```

### 3.3 乐观更新

```typescript
// 乐观更新：先更新 UI，再发送请求
const useUpdateRoom = () => {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: ({ roomId, data }) => roomApi.update(roomId, data),

    // 乐观更新
    onMutate: async (variables) => {
      // 取消正在进行的查询
      await queryClient.cancelQueries({ queryKey: ['rooms', variables.roomId] });

      // 保存之前的值
      const previousRoom = queryClient.getQueryData(['rooms', variables.roomId]);

      // 乐观更新
      queryClient.setQueryData(['rooms', variables.roomId], (old) => ({
        ...old,
        ...variables.data,
      }));

      return { previousRoom };
    },

    // 回滚
    onError: (err, variables, context) => {
      queryClient.setQueryData(
        ['rooms', variables.roomId],
        context?.previousRoom
      );
      message.error('保存失败');
    },

    // 完成后重新获取
    onSettled: (data, error, variables) => {
      queryClient.invalidateQueries({ queryKey: ['rooms', variables.roomId] });
    },
  });
};
```

## 4. 错误处理

### 4.1 全局错误处理

```typescript
// src/services/errorHandler.ts
import { message } from 'antd';
import { AxiosError } from 'axios';

export class ApiError extends Error {
  status: number;
  errors: Record<string, string[]>;

  constructor(status: number, message: string, errors: Record<string, string[]>) {
    super(message);
    this.status = status;
    this.errors = errors;
  }
}

export const handleApiError = (error: unknown) => {
  if (error instanceof AxiosError) {
    const status = error.response?.status;
    const data = error.response?.data;

    switch (status) {
      case 400:
        // 参数校验错误
        if (data?.errors) {
          Object.values(data.errors).forEach((messages: string[]) => {
            messages.forEach((msg) => message.error(msg));
          });
        } else {
          message.error(data?.message || '请求参数错误');
        }
        break;

      case 401:
        // 未认证或 Token 过期
        message.error('登录已过期，请重新登录');
        window.location.href = '/login';
        break;

      case 403:
        // 权限不足
        message.error('权限不足');
        break;

      case 404:
        // 资源不存在
        message.error('请求的资源不存在');
        break;

      case 409:
        // 冲突
        message.error(data?.message || '数据冲突');
        break;

      case 429:
        // 请求过于频繁
        message.error('请求过于频繁，请稍后重试');
        break;

      case 500:
        // 服务器内部错误
        message.error('服务器内部错误，请稍后重试');
        break;

      default:
        message.error('网络错误');
    }

    throw new ApiError(
      status || 500,
      data?.message || 'Unknown error',
      data?.errors || {}
    );
  }

  throw error;
};
```

### 4.2 React Query 错误边界

```typescript
// src/components/ErrorBoundary.tsx
import React, { Component, ErrorInfo, ReactNode } from 'react';
import { Result, Button } from 'antd';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  public state: State = {
    hasError: false,
    error: null,
  };

  public static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  public componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('Uncaught error:', error, errorInfo);
  }

  public render() {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback;
      }

      return (
        <Result
          status="error"
          title="页面错误"
          subTitle="抱歉，页面出现了错误"
          extra={[
            <Button
              type="primary"
              key="reload"
              onClick={() => window.location.reload()}
            >
              刷新页面
            </Button>,
          ]}
        />
      );
    }

    return this.props.children;
  }
}

// 使用
const App = () => (
  <ErrorBoundary>
    <QueryClientProvider client={queryClient}>
      <RouterProvider router={router} />
    </QueryClientProvider>
  </ErrorBoundary>
);
```

```typescript
// src/hooks/useDebounce.ts
import { useState, useEffect } from 'react';

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

// 使用示例：搜索防抖
const SearchComponent = () => {
  const [keyword, setKeyword] = useState('');
  const debouncedKeyword = useDebounce(keyword, 300); // 300ms 防抖

  const { data: results } = useQuery({

  });

      value={keyword}
      onChange={(e) => setKeyword(e.target.value)}

    />
```

```typescript
      {hasNextPage && (
        <Button
          onClick={() => fetchNextPage()}
          loading={isFetchingNextPage}
        >
          加载更多
        </Button>
      )}
    </div>
  );
};
```

```typescript
// src/services/api.ts
api.interceptors.request.use((config) => {
  if (process.env.NODE_ENV === 'development') {
    console.log(`[API Request] ${config.method?.toUpperCase()} ${config.url}`, config);
  }
  return config;
});

api.interceptors.response.use(
  (response) => {
    if (process.env.NODE_ENV === 'development') {
      console.log(`[API Response] ${response.config.url}`, response.data);
    }
    return response;
  },
  (error) => {
    if (process.env.NODE_ENV === 'development') {
      console.error(`[API Error] ${error.config?.url}`, error.response?.data);
    }
    return Promise.reject(error);
  }
);
```
