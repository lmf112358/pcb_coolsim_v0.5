**颜色映射逻辑：**
```typescript
// 冷量热力图（PRD: F3-017）
const getHeatmapColor = (loadValue: number, minLoad: number, maxLoad: number): string => {
  const ratio = (loadValue - minLoad) / (maxLoad - minLoad);
  // 蓝 → 红连续渐变
  const r = Math.round(255 * ratio);
  const g = Math.round(100 * (1 - Math.abs(ratio - 0.5) * 2));
  const b = Math.round(255 * (1 - ratio));
  return `rgb(${r}, ${g}, ${b})`;
};

// 工艺要求颜色编码（PRD: F3-015）
const getProcessColor = (room: RoomHeatmapData): string => {
  if (room.tempPrecision === '±1') return '#FF4D4F'; // 红
  if (room.tempPrecision === '±2') return '#FA8C16'; // 橙
  if (room.cleanlinessLevel === 'ISO7') return '#FADB14'; // 黄
  if (room.cleanlinessLevel === 'ISO8') return '#52C41A'; // 绿
  if (room.cleanlinessLevel === 'ISO7.5') return '#1890FF'; // 蓝
  if (room.pressureDiff && room.pressureDiff > 0) return '#722ED1'; // 紫
  return '#BFBFBF'; // 灰
};
```

**数据下采样（LTTB 算法）：**
```typescript
// LTTB (Largest Triangle Three Buckets) 下采样
const downsampleLTTB = (data: HourlyDataPoint[], targetPoints: number): HourlyDataPoint[] => {
  if (data.length <= targetPoints) return data;

  const sampled: HourlyDataPoint[] = [data[0]]; // 保留首点
  const bucketSize = (data.length - 2) / (targetPoints - 2);

  for (let i = 1; i < targetPoints - 1; i++) {
    const bucketStart = Math.floor((i - 1) * bucketSize) + 1;
    const bucketEnd = Math.min(Math.floor(i * bucketSize) + 1, data.length - 1);

    // 计算下个 bucket 的平均点
    const nextBucketStart = Math.floor(i * bucketSize) + 1;
    const nextBucketEnd = Math.min(Math.floor((i + 1) * bucketSize) + 1, data.length - 1);
    let avgX = 0, avgY = 0;
    for (let j = nextBucketStart; j < nextBucketEnd; j++) {
      avgX += new Date(data[j].timestamp).getTime();
      avgY += data[j].totalLoadKw;
    }
    avgX /= (nextBucketEnd - nextBucketStart);
    avgY /= (nextBucketEnd - nextBucketStart);

    // 选择最大三角形面积的点
    let maxArea = -1;
    let selectedIdx = bucketStart;
    const prevPoint = sampled[sampled.length - 1];
    const prevX = new Date(prevPoint.timestamp).getTime();
    const prevY = prevPoint.totalLoadKw;

    for (let j = bucketStart; j < bucketEnd; j++) {
      const x = new Date(data[j].timestamp).getTime();
      const y = data[j].totalLoadKw;
      const area = Math.abs(
        (prevX - avgX) * (y - prevY) - (prevX - x) * (avgY - prevY)
      ) * 0.5;

      if (area > maxArea) {
        maxArea = area;
        selectedIdx = j;
      }
    }

    sampled.push(data[selectedIdx]);
  }

  sampled.push(data[data.length - 1]); // 保留末点
  return sampled;
};
```
