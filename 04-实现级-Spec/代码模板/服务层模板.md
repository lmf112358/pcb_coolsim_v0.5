## 1. 服务层架构原则

### 1.1 设计原则

1. **单一职责**：每个服务类只负责一个业务领域
2. **依赖注入**：通过构造函数注入依赖
3. **接口隔离**：定义清晰的接口抽象
4. **错误处理**：统一的异常处理机制
5. **日志记录**：关键操作记录日志

# backend/apps/calculations/services/base.py

import logging
from abc import ABC, abstractmethod
from typing import Any, Dict, Optional

from django.utils import timezone

from apps.calculations.exceptions import CalculationError


logger = logging.getLogger(__name__)


class BaseService(ABC):
    """
    基础服务类

    所有业务服务都应继承此类，提供通用功能：
    - 日志记录
    - 异常处理
    - 性能监控
    """

    def __init__(self):
        self.logger = logging.getLogger(self.__class__.__name__)
        self._start_time = None

    def start_timer(self):
        """开始计时"""
        self._start_time = timezone.now()

    def get_elapsed_time(self) -> Optional[float]:
        """获取已用时间（秒）"""
        if self._start_time:
            return (timezone.now() - self._start_time).total_seconds()
        return None

    def log_operation(self, operation: str, success: bool = True, **kwargs):
        """记录操作日志"""
        elapsed = self.get_elapsed_time()
        status = "成功" if success else "失败"

        message = f"{operation} {status}"
        if elapsed:
            message += f"，耗时: {elapsed:.2f}秒"

        if kwargs:
            extra_info = ", ".join(f"{k}={v}" for k, v in kwargs.items())
            message += f"，{extra_info}"

        if success:
            self.logger.info(message)
        else:
            self.logger.error(message)

    def handle_error(self, error: Exception, context: str = "") -> None:
        """
        统一错误处理

        Args:
            error: 原始异常
            context: 错误上下文描述
        """
        error_message = f"{context}: {str(error)}" if context else str(error)
        self.logger.error(error_message, exc_info=True)
        raise CalculationError(error_message) from error

    @abstractmethod
    def execute(self, **kwargs) -> Any:
        """
        执行服务逻辑

        子类必须实现此方法
        """
        pass
```

# backend/apps/calculations/services/static_load.py

from decimal import Decimal
from typing import Dict, List, Optional

from apps.calculations.exceptions import ValidationError
from apps.calculations.services.base import BaseService
from apps.projects.models import Room


class StaticLoadCalculator(BaseService):
    """
    静态负荷计算器

    PRD 引用：F4-001, F4-002, F4-003
    Spec 引用：模块-04-静态冷量计算.md Section 3
    算法伪代码：算法伪代码.md Section 1

    实现 6 步流水线算法计算静态冷负荷
    """

    def __init__(self, room: Room, weather_data: Dict):
        """
        初始化计算器

        Args:
            room: 房间对象
            weather_data: 设计日气象数据
        """
        super().__init__()
        self.room = room
        self.weather_data = weather_data

    def execute(self) -> Dict:
        """
        执行静态负荷计算

        Returns:
            计算结果字典，包含：
            - equipment_heat: 设备散热量
            - lighting_heat: 照明散热量
            - occupancy_heat: 人员散热量
            - envelope_load: 围护结构负荷
            - fresh_air_load: 新风负荷
            - total_load: 总负荷
            - load_per_area: 单位面积负荷
        """
        self.start_timer()

        try:
            # 步骤1：内热源计算
            internal_heat = self._calculate_internal_heat_sources()

            # 步骤2：围护结构计算
            envelope_load = self._calculate_envelope_load()

            # 步骤3：通风负荷计算
            ventilation_load = self._calculate_ventilation_load()

            # 步骤4：工艺负荷计算
            process_load = self._calculate_process_load()

            # 步骤5：安全系数应用
            subtotal = (
                internal_heat['total']
                + envelope_load['total']
                + ventilation_load['total']
                + process_load['total']
            )
            adjusted_load = self._apply_safety_factor(subtotal)

            # 步骤6：汇总
            result = {
                'room_id': self.room.id,
                'room_name': self.room.room_name,
                'equipment_heat': internal_heat['equipment'],
                'lighting_heat': internal_heat['lighting'],
                'occupancy_heat': internal_heat['occupancy'],
                'envelope_load': envelope_load['total'],
                'fresh_air_load': ventilation_load['total'],
                'process_load': process_load['total'],
                'total_load': adjusted_load,
                'load_per_area': adjusted_load / float(self.room.area) if self.room.area > 0 else 0,
            }

            self.log_operation("静态负荷计算", success=True, room_id=self.room.id)
            return result

        except Exception as e:
            self.handle_error(e, f"静态负荷计算失败 (房间 {self.room.id})")

    def _calculate_internal_heat_sources(self) -> Dict[str, Decimal]:
        """
        步骤1：计算内热源散热量

        包括：设备、照明、人员散热
        """
        # 设备散热
        equipment_heat = (
            Decimal(str(self.room.electric_power))
            * Decimal(str(self.room.electric_coeff))
        )

        # 照明散热
        lighting_heat = (
            Decimal(str(self.room.area))
            * Decimal(str(self.room.lighting_index))
        )

        # 人员散热（假设默认值，实际应从房间参数获取）
        occupancy_count = getattr(self.room, 'occupancy_count', 0) or 0
        occupancy_heat_per_person = Decimal('120')  # W/人，可配置
        occupancy_coeff = Decimal('0.8')

        occupancy_heat = (
            Decimal(str(occupancy_count))
            * occupancy_heat_per_person
            * occupancy_coeff
        )

        return {
            'equipment': equipment_heat,
            'lighting': lighting_heat,
            'occupancy': occupancy_heat,
            'total': equipment_heat + lighting_heat + occupancy_heat,
        }

    def _calculate_envelope_load(self) -> Dict[str, Decimal]:
        """
        步骤2：计算围护结构负荷

        包括：墙体传热、屋顶传热、太阳辐射得热
        """
        delta_t = Decimal(str(self.weather_data['outdoor_temp'])) - self.room.indoor_temp

        # 围护结构传热
        construction_load = (
            Decimal(str(self.room.area))
            * Decimal(str(self.room.construction_load_index))
            * delta_t
        )

        # 太阳辐射得热（简化计算）
        solar_radiation = Decimal(str(self.weather_data.get('solar_radiation', 0)))
        solar_gain = solar_radiation * Decimal('0.1') * self.room.area  # 简化系数

        return {
            'conduction': construction_load,
            'solar': solar_gain,
            'total': construction_load + solar_gain,
        }

    def _calculate_ventilation_load(self) -> Dict[str, Decimal]:
        """
        步骤3：计算通风负荷

        包括：新风显热和潜热
        """
        fresh_air_volume = Decimal(str(self.room.fresh_air_volume))

        if fresh_air_volume <= 0:
            return {'sensible': Decimal('0'), 'latent': Decimal('0'), 'total': Decimal('0')}

        # 室内外焓值差
        outdoor_enthalpy = self._calculate_enthalpy(
            self.weather_data['outdoor_temp'],
            self.weather_data['outdoor_humidity']
        )
        indoor_enthalpy = self._calculate_enthalpy(
            float(self.room.indoor_temp),
            float(self.room.indoor_humidity)
        )

        # 质量流量 (kg/s)
        air_density = Decimal('1.2')  # kg/m³
        mass_flow = fresh_air_volume / Decimal('3600') * air_density

        # 新风负荷 (W)
        fresh_air_load = mass_flow * (indoor_enthalpy - outdoor_enthalpy) * Decimal('1000')

        return {
            'sensible': fresh_air_load * Decimal('0.7'),  # 显热占比
            'latent': fresh_air_load * Decimal('0.3'),    # 潜热占比
            'total': fresh_air_load,
        }

    def _calculate_process_load(self) -> Dict[str, Decimal]:
        """
        步骤4：计算工艺负荷

        包括：排风负荷、压差维持负荷
        """
        # 简化计算，实际应根据排风类型分别计算
        return {
            'exhaust': Decimal('0'),
            'pressure': Decimal('0'),
            'total': Decimal('0'),
        }

    def _apply_safety_factor(self, subtotal: Decimal) -> Decimal:
        """
        步骤5：应用安全系数

        Args:
            subtotal: 未调整负荷

        Returns:
            调整后负荷
        """
        confidence_factor = Decimal('1.05')  # 可配置
        adjustment_factor = Decimal('1.0')   # 可配置

        return subtotal * confidence_factor * adjustment_factor

    def _calculate_enthalpy(self, temp: float, humidity: float) -> Decimal:
        """
        计算湿空气焓值

        使用 psychrolib 库或简化公式
        """
        # 简化公式（实际应使用 psychrolib）
        enthalpy = 1.005 * temp + humidity * (2501 + 1.86 * temp) / 1000
        return Decimal(str(enthalpy))

# backend/apps/calculations/services/dynamic_simulation.py

from typing import Dict, List, Optional

from apps.calculations.exceptions import SimulationError
from apps.calculations.services.base import BaseService
from apps.projects.models import Project
from apps.weather.models import WeatherData


class DynamicSimulationService(BaseService):
    """
    动态仿真服务

    PRD 引用：F6-001, F6-002, F6-003
    Spec 引用：模块-06-动态仿真负荷计算.md Section 4
    算法伪代码：算法伪代码.md Section 2

    支持两种仿真方法：
    - 气象数据驱动（方法1）
    - 比例系数法（方法2）
    """

    def __init__(self, project: Project, method: str = 'weather_driven'):
        """
        初始化仿真服务

        Args:
            project: 项目对象
            method: 仿真方法 ('weather_driven' 或 'coefficient')
        """
        super().__init__()
        self.project = project
        self.method = method
        self.rooms = self._load_rooms()
        self.weather_data = self._load_weather_data()

    def execute(
        self,
        start_year: int,
        end_year: int,
        task_id: Optional[int] = None
    ) -> Dict:
        """
        执行动态仿真

        Args:
            start_year: 开始年份
            end_year: 结束年份
            task_id: 仿真任务ID（用于更新进度）

        Returns:
            仿真结果字典
        """
        self.start_timer()

        try:
            if self.method == 'weather_driven':
                results = self._weather_driven_simulation(start_year, end_year, task_id)
            elif self.method == 'coefficient':
                results = self._coefficient_method_simulation(start_year, end_year, task_id)
            else:
                raise SimulationError(f"不支持的仿真方法: {self.method}")

            # 存储结果
            self._store_results(results)

            self.log_operation(
                "动态仿真",
                success=True,
                project_id=self.project.id,
                method=self.method,
                years=f"{start_year}-{end_year}"
            )

            return results

        except Exception as e:
            self.handle_error(e, f"动态仿真失败 (项目 {self.project.id})")

    def _weather_driven_simulation(
        self,
        start_year: int,
        end_year: int,
        task_id: Optional[int]
    ) -> Dict:
        """
        气象数据驱动仿真

        算法参考：算法伪代码.md Section 2.1
        """
        results = {}

        total_hours = (end_year - start_year + 1) * 8760
        processed_hours = 0

        for year in range(start_year, end_year + 1):
            year_results = []

            for hour in range(8760):
                weather = self._get_weather_at_hour(year, hour)
                hour_result = self._simulate_hour(weather, hour)
                year_results.append(hour_result)

                processed_hours += 1

                # 更新进度
                if task_id and processed_hours % 100 == 0:
                    progress = int(processed_hours / total_hours * 100)
                    self._update_task_progress(task_id, progress)

            results[year] = year_results

        return results

    def _coefficient_method_simulation(
        self,
        start_year: int,
        end_year: int,
        task_id: Optional[int]
    ) -> Dict:
        """
        比例系数法仿真

        算法参考：算法伪代码.md Section 2.2
        """
        # 获取静态计算结果
        static_results = self._load_static_results()

        results = {}
        total_hours = (end_year - start_year + 1) * 8760
        processed_hours = 0

        for year in range(start_year, end_year + 1):
            year_results = []

            for hour in range(8760):
                weather = self._get_weather_at_hour(year, hour)
                hour_result = self._apply_coefficients(static_results, weather, hour)
                year_results.append(hour_result)

                processed_hours += 1

                # 更新进度
                if task_id and processed_hours % 100 == 0:
                    progress = int(processed_hours / total_hours * 100)
                    self._update_task_progress(task_id, progress)

            results[year] = year_results

        return results

    def _simulate_hour(self, weather: Dict, hour: int) -> Dict:
        """模拟单个小时的负荷"""
        hour_result = {
            'hour': hour,
            'rooms': []
        }

        for room in self.rooms:
            fresh_air_load = self._calculate_fresh_air_load(room, weather)
            terminal_load = self._calculate_terminal_load(room, weather, hour)
            total_load = fresh_air_load + terminal_load

            # 负荷截断：负负荷处理为 0
            if total_load < 0:
                total_load = 0
                fresh_air_load = 0
                terminal_load = 0

            hour_result['rooms'].append({
                'room_id': room.id,
                'fresh_air_load': fresh_air_load,
                'terminal_load': terminal_load,
                'total_load': total_load,
            })

        return hour_result

    def _load_rooms(self) -> List:
        """加载项目所有房间"""
        return list(
            self.project.buildings
            .filter(is_deleted=False)
            .prefetch_related('floors__rooms')

        )

    def _load_weather_data(self) -> Dict:

        ).order_by('year', 'hour')

    def _get_weather_at_hour(self, year: int, hour: int) -> Dict:

# backend/apps/calculations/services/aggregation.py

from datetime import datetime
from typing import Dict, List, Optional
from decimal import Decimal

from apps.calculations.exceptions import AggregationError
from apps.calculations.services.base import BaseService


class DataAggregationService(BaseService):
    """
    数据聚合服务

    PRD 引用：F7-001, F7-002, F7-003
    Spec 引用：模块-07-8760可视化与基础验证.md Section 3
    算法伪代码：算法伪代码.md Section 3

    负责 8760h 数据的时间聚合和统计计算
    """

    def __init__(self, simulation_id: str):
        """
        初始化聚合服务

        Args:
            simulation_id: 仿真任务ID
        """
        super().__init__()
        self.simulation_id = simulation_id

    def execute(
        self,
        room_id: Optional[int] = None,
        granularity: str = 'hourly',
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None
    ) -> Dict:
        """
        执行数据聚合

        Args:
            room_id: 房间ID（可选，不指定则聚合所有房间）
            granularity: 时间粒度 ('hourly', 'daily', 'monthly')
            start_date: 开始日期（可选）
            end_date: 结束日期（可选）

        Returns:
            聚合后的数据和统计信息
        """
        self.start_timer()

        try:
            # 加载原始数据
            raw_data = self._load_raw_data(room_id, start_date, end_date)

            # 聚合数据
            if granularity == 'daily':
                aggregated_data = self._aggregate_to_daily(raw_data)
            elif granularity == 'monthly':
                aggregated_data = self._aggregate_to_monthly(raw_data)
            else:
                aggregated_data = raw_data

            # 计算统计信息
            statistics = self._calculate_statistics(aggregated_data)

            result = {
                'simulation_id': self.simulation_id,
                'room_id': room_id,
                'granularity': granularity,
                'data': aggregated_data,
                'statistics': statistics,
            }

            self.log_operation(
                "数据聚合",
                success=True,
                granularity=granularity,
                data_count=len(aggregated_data)
            )

            return result

        except Exception as e:
            self.handle_error(e, "数据聚合失败")

    def _load_raw_data(
        self,
        room_id: Optional[int],
        start_date: Optional[datetime],
        end_date: Optional[datetime]
    ) -> List[Dict]:
        """加载原始逐时数据"""
        # 实现从 TimescaleDB 加载
        pass

    def _aggregate_to_daily(self, hourly_data: List[Dict]) -> List[Dict]:
        """
        日聚合

        算法参考：算法伪代码.md Section 3.2
        """
        daily_groups = {}

        for record in hourly_data:
            date_key = record['datetime'].date()

            if date_key not in daily_groups:
                daily_groups[date_key] = {
                    'date': date_key,
                    'loads': [],
                    'outdoor_temps': [],
                }

            daily_groups[date_key]['loads'].append(record['total_load'])
            daily_groups[date_key]['outdoor_temps'].append(record['outdoor_temp'])

        daily_data = []
        for date_key, group in sorted(daily_groups.items()):
            daily_data.append({
                'date': date_key,
                'total_load': sum(group['loads']),
                'avg_load': sum(group['loads']) / len(group['loads']),
                'max_load': max(group['loads']),
                'min_load': min(group['loads']),
                'outdoor_temp_avg': sum(group['outdoor_temps']) / len(group['outdoor_temps']),
                'outdoor_temp_max': max(group['outdoor_temps']),
                'outdoor_temp_min': min(group['outdoor_temps']),
            })

        return daily_data

    def _aggregate_to_monthly(self, hourly_data: List[Dict]) -> List[Dict]:
        """
        月聚合

        算法参考：算法伪代码.md Section 3.2
        """
        monthly_groups = {}

        for record in hourly_data:
            month_key = record['datetime'].strftime('%Y-%m')

            if month_key not in monthly_groups:
                monthly_groups[month_key] = {
                    'month': month_key,
                    'loads': [],
                    'outdoor_temps': [],
                }

            monthly_groups[month_key]['loads'].append(record['total_load'])
            monthly_groups[month_key]['outdoor_temps'].append(record['outdoor_temp'])

        monthly_data = []
        for month_key, group in sorted(monthly_groups.items()):
            monthly_data.append({
                'month': month_key,
                'total_load': sum(group['loads']),
                'avg_load': sum(group['loads']) / len(group['loads']),
                'max_load': max(group['loads']),
                'min_load': min(group['loads']),
                'outdoor_temp_avg': sum(group['outdoor_temps']) / len(group['outdoor_temps']),
            })

        return monthly_data

    def _calculate_statistics(self, data: List[Dict]) -> Dict:
        """
        计算统计信息

        算法参考：算法伪代码.md Section 3.3
        """
        if not data:
            return {}

        loads = [record.get('total_load', record.get('avg_load', 0)) for record in data]

        peak_load = max(loads)
        valley_load = min(loads)
        avg_load = sum(loads) / len(loads)
        total_energy = sum(loads) / 1000  # W -> kWh

        # 计算百分位数
        percentiles = self._calculate_percentiles(loads, [10, 25, 50, 75, 90, 95, 99])

        return {
            'peak_load': peak_load,
            'valley_load': valley_load,
            'avg_load': avg_load,
            'total_energy_kwh': total_energy,
            'load_factor': avg_load / peak_load if peak_load > 0 else 0,
            'percentiles': percentiles,
        }

    def _calculate_percentiles(
        self,
        data: List[float],
        percentiles: List[int]
    ) -> Dict[int, float]:
        """计算百分位数"""
        sorted_data = sorted(data)
        n = len(sorted_data)
        result = {}

        for p in percentiles:
            index = (p / 100) * (n - 1)
            lower_index = int(index)
            upper_index = min(lower_index + 1, n - 1)
            fraction = index - lower_index

            value = sorted_data[lower_index] * (1 - fraction) + sorted_data[upper_index] * fraction
            result[p] = value

        return result
```

## 6. LTTB 下采样服务模板

```python
# backend/apps/calculations/services/downsampling.py

import math
from typing import List, Tuple

from apps.calculations.services.base import BaseService


class LTTBDownsampler(BaseService):
    """
    LTTB 下采样服务

    算法伪代码：算法伪代码.md Section 4

    实现最大三角形三桶算法（Largest Triangle Three Buckets）
    用于将大量数据点下采样到目标数量，保持数据形状
    """

    def __init__(self, target_count: int = 1000):
        """
        初始化下采样器

        Args:
            target_count: 目标数据点数量
        """
        super().__init__()
        self.target_count = target_count

    def execute(self, data: List[Tuple[float, float]]) -> List[Tuple[float, float]]:
        """
        执行 LTTB 下采样

        Args:
            data: 原始数据点列表 [(x, y), ...]

        Returns:
            下采样后的数据点列表
        """
        self.start_timer()

        n = len(data)

        # 数据量不足，直接返回
        if n <= self.target_count:
            return data

        sampled = [data[0]]  # 保留首点

        bucket_size = (n - 2) / (self.target_count - 2)

        for i in range(1, self.target_count - 1):
            # 计算下一个 bucket 的平均点
            bucket_start = int((i) * bucket_size) + 1
            bucket_end = min(int((i + 1) * bucket_size) + 1, n - 1)

            avg_x = 0
            avg_y = 0
            for j in range(bucket_start, bucket_end):
                avg_x += data[j][0]
                avg_y += data[j][1]
            avg_x /= (bucket_end - bucket_start)
            avg_y /= (bucket_end - bucket_start)

            # 计算当前 bucket 的范围
            current_start = int((i - 1) * bucket_size) + 1
            current_end = min(int(i * bucket_size) + 1, n)

            # 在当前 bucket 中找到最大三角形面积的点
            max_area = -1
            selected_point = current_start
            prev_point = sampled[-1]

            for j in range(current_start, current_end):
                area = self._calculate_triangle_area(
                    prev_point[0], prev_point[1],
                    data[j][0], data[j][1],
                    avg_x, avg_y
                )

                if area > max_area:
                    max_area = area
                    selected_point = j

            sampled.append(data[selected_point])

        sampled.append(data[-1])  # 保留末点

        self.log_operation(
            "LTTB下采样",
            success=True,
            original_count=n,
            sampled_count=len(sampled)
        )

        return sampled

    def _calculate_triangle_area(
        self,
        x1: float, y1: float,
        x2: float, y2: float,
        x3: float, y3: float
    ) -> float:
        """
        计算三角形面积（向量叉积法）

        算法参考：算法伪代码.md Section 4
        """
        cross_product = abs((x1 - x3) * (y2 - y1) - (x1 - x2) * (y3 - y1))
        return cross_product * 0.5

    def auto_downsample(self, data: List[Tuple[float, float]]) -> List[Tuple[float, float]]:
        """
        自动选择下采样策略

        算法参考：算法伪代码.md Section 4 - DownsampleStrategy
        """
        n = len(data)

        if n <= 500:
            return data
        elif n <= 2000:
            return self.execute_with_target(data, min(self.target_count, 500))
        elif n <= 10000:
            return self.execute_with_target(data, min(self.target_count, 1000))
        else:
            return self.execute_with_target(data, min(self.target_count, 5000))

    def execute_with_target(
        self,
        data: List[Tuple[float, float]],
        target: int
    ) -> List[Tuple[float, float]]:
        """使用指定目标数量执行下采样"""
        original_target = self.target_count
        self.target_count = target
        result = self.execute(data)
        self.target_count = original_target
        return result
```

from celery import shared_task
from django.utils import timezone

from apps.calculations.models import CalculationTask, SimulationTask

from apps.calculations.services.dynamic_simulation import DynamicSimulationService


@shared_task(bind=True, max_retries=3)
def calculate_static_load_task(self, task_id: int, room_id: int, weather_data: dict):
    """
    异步执行静态负荷计算

    Args:
        task_id: 计算任务ID
        room_id: 房间ID
        weather_data: 气象数据
    """
    try:
        # 更新任务状态
        task = CalculationTask.objects.get(id=task_id)
        task.status = CalculationTask.Status.PROCESSING
        task.started_at = timezone.now()
        task.save(update_fields=['status', 'started_at'])

        # 加载房间数据
        from apps.projects.models import Room
        room = Room.objects.get(id=room_id)

        # 执行计算
        calculator = StaticLoadCalculator(room, weather_data)
        result = calculator.execute()

        # 保存结果
        from apps.calculations.models import StaticLoadResult
        StaticLoadResult.objects.create(
            room=room,
            task=task,
            equipment_heat=result['equipment_heat'],
            lighting_heat=result['lighting_heat'],
            occupancy_heat=result['occupancy_heat'],
            envelope_load=result['envelope_load'],
            fresh_air_load=result['fresh_air_load'],
            total_load=result['total_load'],
            load_per_area=result['load_per_area'],
        )

        # 更新任务状态
        task.status = CalculationTask.Status.COMPLETED
        task.completed_at = timezone.now()
        task.progress = 100
        task.save(update_fields=['status', 'completed_at', 'progress'])

        return {'success': True, 'task_id': task_id}

    except Exception as exc:
        # 更新任务状态为失败
        task = CalculationTask.objects.get(id=task_id)
        task.status = CalculationTask.Status.FAILED
        task.error_message = str(exc)
        task.save(update_fields=['status', 'error_message'])

        # 重试
        raise self.retry(exc=exc, countdown=60)
