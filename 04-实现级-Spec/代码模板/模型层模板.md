```python
# backend/apps/common/models.py

from django.db import models
from django.utils import timezone


class TimestampedModel(models.Model):
    """
    带时间戳的基础模型

    所有业务模型都应继承此类，自动添加 created_at 和 updated_at 字段
    """

    created_at = models.DateTimeField(
        auto_now_add=True,
        verbose_name="创建时间",
        help_text="记录创建时间"
    )
    updated_at = models.DateTimeField(
        auto_now=True,
        verbose_name="更新时间",
        help_text="最后更新时间"
    )

    class Meta:
        abstract = True
        ordering = ['-created_at']

    def save(self, *args, **kwargs):
        """重写 save 方法，确保时间戳更新"""
        self.updated_at = timezone.now()
        super().save(*args, **kwargs)


class SoftDeleteModel(models.Model):
    """
    软删除模型基类

    使用 is_deleted 标记删除状态，不实际删除记录
    """

    is_deleted = models.BooleanField(
        default=False,
        verbose_name="是否删除",
        help_text="软删除标记"
    )
    deleted_at = models.DateTimeField(
        null=True,
        blank=True,
        verbose_name="删除时间",
        help_text="软删除时间"
    )

    class Meta:
        abstract = True

    def delete(self, *args, **kwargs):
        """软删除：标记为已删除"""
        self.is_deleted = True
        self.deleted_at = timezone.now()
        self.save(update_fields=['is_deleted', 'deleted_at'])

    def hard_delete(self, *args, **kwargs):
        """硬删除：实际删除记录"""
        super().delete(*args, **kwargs)

    def restore(self):
        """恢复软删除的记录"""
        self.is_deleted = False
        self.deleted_at = None
        self.save(update_fields=['is_deleted', 'deleted_at'])
```

```python
from django.contrib.auth.models import AbstractUser

    )

        max_length=20,
        blank=True,
        null=True,
        verbose_name="手机号",
        help_text="联系电话"
    )
    company = models.CharField(
        max_length=100,
        blank=True,
        null=True,
        verbose_name="公司",
        help_text="所属公司"
    )
    department = models.CharField(
        max_length=100,
        blank=True,
        null=True,
        verbose_name="部门",
        help_text="所属部门"
    )

    # 配置
    USERNAME_FIELD = 'email'
    REQUIRED_FIELDS = ['username']

    class Meta:
        db_table = 'user'
        verbose_name = '用户'
        verbose_name_plural = '用户'
        ordering = ['-created_at']

    def __str__(self):
        return f"{self.email} ({self.get_role_display()})"

    @property
    def is_admin(self):
        return self.role == self.Role.ADMIN

    @property
    def is_project_manager(self):
        return self.role == self.Role.PROJECT_MANAGER

    @property
    def can_manage_projects(self):
        return self.role in [self.Role.ADMIN, self.Role.PROJECT_MANAGER]
```

```python
from django.conf import settings
from django.db import models

from apps.common.models import TimestampedModel, SoftDeleteModel


class Project(TimestampedModel, SoftDeleteModel):
    """
    项目模型

    PRD 引用：F1-003, F1-004（项目管理）
    Spec 引用：模块-01-账户与项目管理.md Section 3

        DRAFT = 'draft', '草稿'
        ACTIVE = 'active', '进行中'
        COMPLETED = 'completed', '已完成'
        ARCHIVED = 'archived', '已归档'

    owner = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.CASCADE,

        max_length=200,

        decimal_places=2,

        blank=True,
        null=True,

        help_text="室内设计温度"

        max_digits=5,
        decimal_places=1,
        default=55.0,
        verbose_name="室内湿度 (%)",
        help_text="室内设计湿度"

    # 负荷计算参数
    construction_load_index = models.DecimalField(
        max_digits=8,
        decimal_places=2,
        default=0,
        verbose_name="围护结构负荷指标 (W/m²)",
        help_text="围护结构单位面积负荷"

    lighting_index = models.DecimalField(
        max_digits=8,
        decimal_places=2,
        default=0,
        verbose_name="照明负荷指标 (W/m²)",
        help_text="单位面积照明负荷"
    )

    # 设备参数
    electric_power = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        default=0,
        verbose_name="电气设备功率 (W)",

    )
    electric_coeff = models.DecimalField(
        max_digits=4,
        decimal_places=2,
        default=1.0,
        verbose_name="电气设备同时使用系数",
        help_text="电气设备同时使用系数 (0-1)"
    )

    # 新风参数
    fresh_air_volume = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        default=0,

        help_text="新风量"
    )

    # 冷冻水配置
    water_grade = models.CharField(
        max_length=10,
        choices=WaterGrade.choices,
        default=WaterGrade.LOW,
        verbose_name="冷冻水等级",
        help_text="冷冻水等级配置"
    )

    class Meta:
        db_table = 'room'
        verbose_name = '功能区域'
        verbose_name_plural = '功能区域'
        ordering = ['floor', 'room_name']

    def __str__(self):
        return self.room_name

    @property
    def volume(self):
        """房间体积"""
        return float(self.area) * float(self.ceiling_height)
```

## 4. 模型最佳实践

### 4.1 字段顺序规范

```
1. 主键 (id)
2. 外键关系 (ForeignKey, OneToOneField, ManyToManyField)
3. 核心业务字段
4. 配置字段
5. 时间戳字段
6. 软删除字段（如继承 SoftDeleteModel）
```

### 4.2 Meta 类规范

```python
class Meta:
    db_table = 'table_name'           # 明确指定表名
    verbose_name = '中文表名'          # 单数形式
    verbose_name_plural = '中文表名'   # 复数形式
    ordering = ['-created_at']         # 默认排序
    unique_together = ['field1', 'field2']  # 唯一约束
    indexes = [
        models.Index(fields=['field1', 'field2']),  # 复合索引
    ]
```

### 4.3 字段 help_text 规范

所有字段必须提供 `help_text`，说明：
- 字段的业务含义
- 数据单位（如有）
- 取值范围或约束

```python
area = models.DecimalField(
    max_digits=10,
    decimal_places=2,
    verbose_name="面积 (m²)",
    help_text="功能区域面积，单位：平方米"
)
```

### 4.4 属性方法规范

使用 `@property` 装饰器定义计算属性：

```python
@property
def total_area(self):
    """计算总面积"""
    return sum(child.area for child in self.children.all())

@property
def is_admin(self):
    """检查是否为管理员"""
    return self.role == self.Role.ADMIN
```
