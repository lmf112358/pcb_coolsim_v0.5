```typescript
// frontend/src/services/api.ts

import axios, { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import { message } from 'antd';

/**
 * API 客户端配置
 */
const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || '/api/v1';

const apiClient: AxiosInstance = axios.create({
  baseURL: API_BASE_URL,
  timeout: 30000,
  headers: {
    'Content-Type': 'application/json',
  },
});

/**
 * 请求拦截器
 */
apiClient.interceptors.request.use(
  (config) => {
    // 添加认证 token
    const token = localStorage.getItem('access_token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

/**
 * 响应拦截器
 */
apiClient.interceptors.response.use(
  (response: AxiosResponse) => {
    return response;
  },
  (error) => {
    // 处理错误响应
    if (error.response) {
      const { status, data } = error.response;

      switch (status) {
        case 401:
          // Token 过期，跳转登录
          localStorage.removeItem('access_token');
          window.location.href = '/login';
          break;
        case 403:
          message.error('没有权限执行此操作');
          break;
        case 404:
          message.error('请求的资源不存在');
          break;
        case 500:
          message.error('服务器内部错误');
          break;
        default:
          if (data?.message) {
            message.error(data.message);
          }
      }
    } else if (error.request) {
      message.error('网络错误，请检查网络连接');
    }

    return Promise.reject(error);
  }
);

export default apiClient;
```

### 3.2 认证服务

```typescript
// frontend/src/services/authService.ts

import apiClient from './api';
import { ApiResponse, User } from '@/types';

interface LoginRequest {
  email: string;
  password: string;
}

interface LoginResponse {
  access: string;
  refresh: string;
  user: User;
}

interface TokenRefreshRequest {
  refresh: string;
}

interface TokenRefreshResponse {
  access: string;
}

/**
 * 认证服务
 */
export const authService = {
  /**
   * 登录
   */
  async login(credentials: LoginRequest): Promise<LoginResponse> {
    const response = await apiClient.post<ApiResponse<LoginResponse>>(
      '/auth/login/',
      credentials
    );

    const { access, refresh, user } = response.data.data;

    // 存储 token
    localStorage.setItem('access_token', access);
    localStorage.setItem('refresh_token', refresh);

    return response.data.data;
  },

  /**
   * 登出
   */
  async logout(): Promise<void> {
    const refresh = localStorage.getItem('refresh_token');

    if (refresh) {
      try {
        await apiClient.post('/auth/logout/', { refresh });
      } catch (error) {
        console.error('Logout error:', error);
      }
    }

    // 清除本地存储
    localStorage.removeItem('access_token');
    localStorage.removeItem('refresh_token');
  },

  /**
   * 刷新 Token
   */
  async refreshToken(): Promise<string> {
    const refresh = localStorage.getItem('refresh_token');

    if (!refresh) {
      throw new Error('No refresh token available');
    }

    const response = await apiClient.post<ApiResponse<TokenRefreshResponse>>(
      '/auth/refresh/',
      { refresh }
    );

    const { access } = response.data.data;
    localStorage.setItem('access_token', access);

    return access;
  },

  /**
   * 获取当前用户信息
   */
  async getCurrentUser(): Promise<User> {
    const response = await apiClient.get<ApiResponse<User>>('/auth/user/');
    return response.data.data;
  },

  /**
   * 检查是否已认证
   */
  isAuthenticated(): boolean {
    return !!localStorage.getItem('access_token');
  },
};

export default authService;
```

```typescript
// frontend/src/hooks/useAuth.ts

import { useState, useEffect, useCallback } from 'react';
import { useNavigate } from 'react-router-dom';
import { authService } from '@/services/authService';
import { User } from '@/types';

interface UseAuthReturn {
  user: User | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => Promise<void>;
}

/**
 * 认证 Hook
 */
export const useAuth = (): UseAuthReturn => {
  const [user, setUser] = useState<User | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const navigate = useNavigate();

  useEffect(() => {
    const initAuth = async () => {
      if (authService.isAuthenticated()) {
        try {
          const currentUser = await authService.getCurrentUser();
          setUser(currentUser);
        } catch (error) {
          console.error('Failed to get current user:', error);
          localStorage.removeItem('access_token');
          localStorage.removeItem('refresh_token');
        }
      }
      setIsLoading(false);
    };

    initAuth();
  }, []);

  const login = useCallback(async (email: string, password: string) => {
    setIsLoading(true);
    try {
      const response = await authService.login({ email, password });
      setUser(response.user);
      navigate('/map');
    } catch (error) {
      console.error('Login failed:', error);
      throw error;
    } finally {
      setIsLoading(false);
    }
  }, [navigate]);

  const logout = useCallback(async () => {
    setIsLoading(true);
    try {
      await authService.logout();
      setUser(null);
      navigate('/login');
    } catch (error) {
      console.error('Logout failed:', error);
    } finally {
      setIsLoading(false);
    }
  }, [navigate]);

  return {
    user,
    isAuthenticated: !!user,
    isLoading,
    login,
    logout,
  };
};

export default useAuth;
```

### 4.2 计算任务 Hook

```typescript
// frontend/src/hooks/useSimulation.ts

import { useState, useEffect, useCallback, useRef } from 'react';
import { calculationService } from '@/services/calculationService';
import { TaskStatus, HourlyDataPoint, Statistics } from '@/types';

interface UseSimulationOptions {
  taskId: string;
  autoRefresh?: boolean;
  refreshInterval?: number;
}

interface UseSimulationReturn {
  status: TaskStatus | null;
  data: HourlyDataPoint[] | null;
  statistics: Statistics | null;
  isLoading: boolean;
  error: string | null;
  refresh: () => Promise<void>;
}

/**
 * 仿真任务 Hook

 * 支持自动轮询任务状态和获取结果
 */
export const useSimulation = ({
  taskId,
  autoRefresh = true,
  refreshInterval = 2000,
}: UseSimulationOptions): UseSimulationReturn => {
  const [status, setStatus] = useState<TaskStatus | null>(null);
  const [data, setData] = useState<HourlyDataPoint[] | null>(null);
  const [statistics, setStatistics] = useState<Statistics | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const intervalRef = useRef<NodeJS.Timeout | null>(null);

  const fetchStatus = useCallback(async () => {
    try {
      const taskStatus = await calculationService.getSimulationStatus(taskId);
      setStatus(taskStatus);

      // 如果任务完成，获取结果
      if (taskStatus.status === 'completed' && !data) {
        const results = await calculationService.getSimulationResults(taskId, {
          granularity: 'hourly',
        });
        setData(results.data);
        setStatistics(results.statistics);
      }

      // 如果任务失败，设置错误
      if (taskStatus.status === 'failed') {
        setError(taskStatus.error_message || '任务执行失败');
      }
    } catch (err) {
      setError(err instanceof Error ? err.message : '获取任务状态失败');
    } finally {
      setIsLoading(false);
    }
  }, [taskId, data]);

  useEffect(() => {
    fetchStatus();

    // 自动轮询
    if (autoRefresh) {
      intervalRef.current = setInterval(() => {
        if (status?.status !== 'completed' && status?.status !== 'failed') {
          fetchStatus();
        } else {
          if (intervalRef.current) {
            clearInterval(intervalRef.current);
          }
        }
      }, refreshInterval);
    }

    return () => {
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
    };
  }, [fetchStatus, autoRefresh, refreshInterval, status?.status]);

  const refresh = useCallback(async () => {
    setIsLoading(true);
    await fetchStatus();
  }, [fetchStatus]);

  return {
    status,
    data,
    statistics,
    isLoading,
    error,
    refresh,
  };
};

export default useSimulation;
```

### 4.3 导出 Hook

```typescript
// frontend/src/hooks/useExport.ts

import { useState, useCallback } from 'react';
import { message } from 'antd';
import apiClient from '@/services/api';
import { ExportTask } from '@/types';

interface UseExportOptions {
  projectId: number;
}

interface UseExportReturn {
  exportTask: ExportTask | null;
  isExporting: boolean;
  error: string | null;
  exportStaticReport: (options?: Record<string, any>) => Promise<void>;
  exportSimulationData: (simulationId: string, options?: Record<string, any>) => Promise<void>;
  downloadFile: () => Promise<void>;
}

/**
 * 导出 Hook
 */
export const useExport = ({ projectId }: UseExportOptions): UseExportReturn => {
  const [exportTask, setExportTask] = useState<ExportTask | null>(null);
  const [isExporting, setIsExporting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const pollTaskStatus = useCallback(async (taskId: string) => {
    const maxAttempts = 60; // 最多轮询 60 次
    let attempts = 0;

    const poll = async () => {
      if (attempts >= maxAttempts) {
        setError('导出超时，请重试');
        setIsExporting(false);
        return;
      }

      try {
        const response = await apiClient.get(`/exports/${taskId}/status/`);
        const task = response.data.data as ExportTask;

        setExportTask(task);

        if (task.status === 'completed') {
          message.success('导出完成，可以下载文件');
          setIsExporting(false);
        } else if (task.status === 'failed') {
          setError(task.error_message || '导出失败');
          setIsExporting(false);
        } else {
          attempts++;
          setTimeout(poll, 2000); // 2 秒后重试
        }
      } catch (err) {
        setError('查询导出状态失败');
        setIsExporting(false);
      }
    };

    await poll();
  }, []);

  const exportStaticReport = useCallback(async (options?: Record<string, any>) => {
    setIsExporting(true);
    setError(null);

    try {
      const response = await apiClient.post(
        `/projects/${projectId}/exports/static-report/`,
        options
      );

      const task = response.data.data as ExportTask;
      setExportTask(task);

      // 开始轮询状态
      await pollTaskStatus(task.task_id);
    } catch (err) {
      setError(err instanceof Error ? err.message : '提交导出任务失败');
      setIsExporting(false);
    }
  }, [projectId, pollTaskStatus]);

  const exportSimulationData = useCallback(async (
    simulationId: string,
    options?: Record<string, any>
  ) => {
    setIsExporting(true);
    setError(null);

    try {
      const response = await apiClient.post(
        `/projects/${projectId}/exports/simulation-data/`,
        {
          simulation_id: simulationId,
          ...options,
        }
      );

      const task = response.data.data as ExportTask;
      setExportTask(task);

      await pollTaskStatus(task.task_id);
    } catch (err) {
      setError(err instanceof Error ? err.message : '提交导出任务失败');
      setIsExporting(false);
    }
  }, [projectId, pollTaskStatus]);

  const downloadFile = useCallback(async () => {
    if (!exportTask?.download_url) {
      setError('没有可下载的文件');
      return;
    }

    try {
      const response = await apiClient.get(exportTask.download_url, {
        responseType: 'blob',
      });

      // 创建下载链接
      const url = window.URL.createObjectURL(new Blob([response.data]));
      const link = document.createElement('a');
      link.href = url;
      link.setAttribute('download', exportTask.file_name || 'export');
      document.body.appendChild(link);
      link.click();
      link.remove();
      window.URL.revokeObjectURL(url);
    } catch (err) {
      setError('下载文件失败');
    }
  }, [exportTask]);

  return {
    exportTask,
    isExporting,
    error,
    exportStaticReport,
    exportSimulationData,
    downloadFile,
  };
};

export default useExport;
```

```typescript
// frontend/src/components/common/DownloadButton/index.tsx

import React from 'react';
import { Button, Tooltip } from 'antd';
import { DownloadOutlined, LoadingOutlined } from '@ant-design/icons';

interface DownloadButtonProps {
  url: string;
  fileName?: string;
  loading?: boolean;
  disabled?: boolean;
  children?: React.ReactNode;
  onDownloadStart?: () => void;
  onDownloadComplete?: () => void;
  onError?: (error: Error) => void;
}

/**
 * 下载按钮组件
 */
const DownloadButton: React.FC<DownloadButtonProps> = ({
  url,
  fileName,
  loading = false,
  disabled = false,
  children,
  onDownloadStart,
  onDownloadComplete,
  onError,
}) => {
  const handleDownload = async () => {
    try {
      onDownloadStart?.();

      const response = await fetch(url);
      if (!response.ok) {
        throw new Error('下载失败');
      }

      const blob = await response.blob();
      const downloadUrl = window.URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = downloadUrl;
      link.setAttribute('download', fileName || 'download');
      document.body.appendChild(link);
      link.click();
      link.remove();
      window.URL.revokeObjectURL(downloadUrl);

      onDownloadComplete?.();
    } catch (error) {
      onError?.(error instanceof Error ? error : new Error('下载失败'));
    }
  };

  return (
    <Tooltip title={disabled ? '无法下载' : '点击下载'}>
      <Button
        type="primary"
        icon={loading ? <LoadingOutlined /> : <DownloadOutlined />}
        loading={loading}
        disabled={disabled}
        onClick={handleDownload}
      >
        {children || '下载'}
      </Button>
    </Tooltip>
  );
};

export default DownloadButton;
```

```typescript
    <div>
      {label && <div style={{ marginBottom: 8 }}>{label}</div>}
      <Progress
        percent={progress}
        status={getStatusColor()}
        showInfo={showPercent}
        size={size === 'small' ? 'small' : 'default'}
```

```typescript
// frontend/src/components/ErrorBoundary.tsx

import React, { Component, ErrorInfo, ReactNode } from 'react';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('Uncaught error:', error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div style={{ padding: 20, textAlign: 'center' }}>
          <h2>出现错误</h2>
          <p>{this.state.error?.message}</p>
          <button onClick={() => this.setState({ hasError: false, error: null })}>
            重试
          </button>
        </div>
      );
    }

    return this.props.children;
  }
}

export default ErrorBoundary;
```
