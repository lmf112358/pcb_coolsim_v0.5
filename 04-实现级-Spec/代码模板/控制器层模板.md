```python
# backend/apps/common/views.py

from rest_framework import viewsets, status
from rest_framework.decorators import action
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated
from django.utils import timezone

from apps.common.permissions import IsOwnerOrReadOnly


class BaseModelViewSet(viewsets.ModelViewSet):
    """
    基础 ViewSet

    提供通用功能：
    - 认证和权限
    - 分页
    - 错误处理
    - 日志记录
    """

    permission_classes = [IsAuthenticated]

    def get_queryset(self):
        """
        根据用户权限过滤查询集

        - 管理员：所有记录
        - 普通用户：仅自己的记录
        """
        queryset = super().get_queryset()

        if self.request.user.is_admin:
            return queryset

        # 默认只返回用户的记录
        if hasattr(queryset.model, 'owner'):
            return queryset.filter(owner=self.request.user)
        elif hasattr(queryset.model, 'user'):
            return queryset.filter(user=self.request.user)

        return queryset

    def perform_create(self, serializer):
        """创建时自动设置用户"""
        if hasattr(serializer.Meta.model, 'owner'):
            serializer.save(owner=self.request.user)
        elif hasattr(serializer.Meta.model, 'user'):
            serializer.save(user=self.request.user)
        else:
            serializer.save()

    def create(self, request, *args, **kwargs):
        """重写创建方法，添加日志"""
        response = super().create(request, *args, **kwargs)

        if response.status_code == status.HTTP_201_CREATED:
            self.log_action('create', response.data)

        return response

    def update(self, request, *args, **kwargs):
        """重写更新方法，添加日志"""
        response = super().update(request, *args, **kwargs)

        if response.status_code == status.HTTP_200_OK:
            self.log_action('update', response.data)

        return response

    def destroy(self, request, *args, **kwargs):
        """重写删除方法，添加日志"""
        instance = self.get_object()
        self.log_action('delete', {'id': instance.id})

        return super().destroy(request, *args, **kwargs)

    def log_action(self, action: str, data: dict):
        """记录操作日志"""
        import logging
        logger = logging.getLogger('audit')
        logger.info(
            f"User {self.request.user.id} performed {action} on {self.basename}",
            extra={
                'user_id': self.request.user.id,
                'action': action,
                'resource': self.basename,
                'data': data,
                'timestamp': timezone.now().isoformat(),
            }
        )
```

```python
from rest_framework import serializers

from apps.projects.models import Project, Building, Floor, Room


class ProjectSerializer(serializers.ModelSerializer):
    """
    项目序列化器

    用于列表展示和创建
    """

    building_count = serializers.IntegerField(read_only=True)
    total_area = serializers.DecimalField(
        max_digits=12,
        decimal_places=2,
        read_only=True
    )

    class Meta:
        model = Project
        fields = [
            'id',
            'name',
            'code',
            'status',
            'description',
            'location',
            'latitude',
            'longitude',
            'building_count',
            'total_area',
            'created_at',
            'updated_at',
        ]
        read_only_fields = ['id', 'created_at', 'updated_at']

    def validate_code(self, value):
        """验证项目编号唯一性"""
        if self.instance and self.instance.code == value:
            return value

        if Project.objects.filter(code=value).exists():
            raise serializers.ValidationError("项目编号已存在")

        return value


class ProjectDetailSerializer(ProjectSerializer):
    """
    项目详情序列化器

    包含关联的建筑列表
    """

    buildings = BuildingSerializer(many=True, read_only=True)

    class Meta(ProjectSerializer.Meta):
        fields = ProjectSerializer.Meta.fields + ['buildings']


class BuildingSerializer(serializers.ModelSerializer):
    """
    建筑序列化器
    """

    floor_count = serializers.IntegerField(read_only=True)

    class Meta:
        model = Building
        fields = [
            'id',
            'project',
            'name',
            'code',
            'description',
            'floor_count',
            'created_at',
            'updated_at',
        ]
        read_only_fields = ['id', 'created_at', 'updated_at']


class FloorSerializer(serializers.ModelSerializer):
    """
    楼层序列化器
    """

    room_count = serializers.IntegerField(read_only=True)

    class Meta:
        model = Floor
        fields = [
            'id',
            'building',
            'name',
            'level',
            'description',
            'room_count',
            'created_at',
            'updated_at',
        ]
        read_only_fields = ['id', 'created_at', 'updated_at']


class RoomSerializer(serializers.ModelSerializer):
    """
    功能区域序列化器

    PRD 引用：F2-001, F2-002, F2-003
    """

    class Meta:
        model = Room
        fields = [
            'id',
            'floor',
            'room_name',
            'room_code',
            'area',
            'ceiling_height',
            'design_temp',
            'design_humidity',
            'indoor_temp',
            'indoor_humidity',
            'construction_load_index',
            'lighting_index',
            'electric_power',
            'electric_coeff',
            'fresh_air_volume',
            'water_grade',
            'created_at',
            'updated_at',
        ]

        """验证面积合理性"""
        if value <= 0:
            raise serializers.ValidationError("面积必须大于 0")
        if value > 100000:
            raise serializers.ValidationError("面积超出合理范围")
        return value

        """验证层高合理性"""
        if value <= 0:
            raise serializers.ValidationError("层高必须大于 0")
        if value > 20:
            raise serializers.ValidationError("层高超出合理范围")
        return value

    def validate_electric_coeff(self, value):
        """验证系数范围"""
        if value < 0 or value > 1:
            raise serializers.ValidationError("系数必须在 0-1 之间")
        return value
```

```python
            room_id=room_id,

        )
```

```python
            open(file_path, 'rb'),
            content_type=task.content_type
        )
        response['Content-Disposition'] = f'attachment; filename="{task.file_name}"'

        return response
```

```python
# backend/apps/common/permissions.py

from rest_framework import permissions


class IsOwnerOrReadOnly(permissions.BasePermission):
    """
    所有者或只读权限

    只有对象的所有者才能修改，其他人只能读取
    """

    def has_object_permission(self, request, view, obj):
        # 读取权限对所有请求开放
        if request.method in permissions.SAFE_METHODS:
            return True

        # 写入权限只对所有者开放
        return obj.owner == request.user


class IsProjectMember(permissions.BasePermission):
    """
    项目成员权限

    只有项目成员才能访问项目资源
    """

    def has_permission(self, request, view):
        """检查是否有项目访问权限"""
        project_id = view.kwargs.get('project_pk') or request.data.get('project_id')

        if not project_id:
            return False

        from apps.projects.models import Project
        try:
            project = Project.objects.get(id=project_id)
        except Project.DoesNotExist:
            return False

        # 管理员可以访问所有项目
        if request.user.is_admin:
            return True

        # 检查是否是项目成员
        return project.members.filter(user=request.user).exists()

    def has_object_permission(self, request, view, obj):
        """检查对象级别的权限"""
        # 读取权限检查项目成员身份
        if request.method in permissions.SAFE_METHODS:
            return self._is_project_member(request.user, obj)

        # 写入权限需要更高权限
        return self._can_edit_project(request.user, obj)

    def _is_project_member(self, user, obj):
        """检查是否是项目成员"""
        if hasattr(obj, 'project'):
            project = obj.project
        elif hasattr(obj, 'id'):
            project = obj
        else:
            return False

        return project.members.filter(user=user).exists()

    def _can_edit_project(self, user, obj):
        """检查是否有编辑权限"""
        if user.is_admin:
            return True

        if hasattr(obj, 'project'):
            project = obj.project
        elif hasattr(obj, 'id'):
            project = obj
        else:
            return False

        # 只有项目所有者和管理员可以编辑
        return project.owner == user


class IsAdminOrProjectManager(permissions.BasePermission):
    """
    管理员或项目经理权限
    """

    def has_permission(self, request, view):
        return (
            request.user.is_admin or
            request.user.is_project_manager
        )
```

## 7. 错误处理模板

```python
# backend/apps/common/exceptions.py

from rest_framework.views import exception_handler
from rest_framework.response import Response
from rest_framework import status


def custom_exception_handler(exc, context):
    """
    自定义异常处理器

    统一错误响应格式
    """
    # 调用默认异常处理器
    response = exception_handler(exc, context)

    if response is not None:
        # 构建统一错误响应
        error_response = {
            'code': response.status_code,
            'message': '请求失败',
            'errors': {},
        }

        if isinstance(response.data, dict):
            if 'detail' in response.data:
                error_response['message'] = response.data['detail']
            else:
                error_response['errors'] = response.data
        elif isinstance(response.data, list):
            error_response['errors'] = {'detail': response.data}

        response.data = error_response

    return response
```

```python
# 创建任务并返回任务ID
def create(self, request, *args, **kwargs):
    serializer = self.get_serializer(data=request.data)
    serializer.is_valid(raise_exception=True)

    # 创建任务记录
    task = MyTask.objects.create(
        user=request.user,
        status='pending',
        parameters=serializer.validated_data,
    )

    # 异步执行
    from apps.myapp.tasks import my_async_task

        {
            'task_id': task.id,
            'status': 'pending',
            'message': '任务已提交'
        },
        status=status.HTTP_202_ACCEPTED
    )
```
