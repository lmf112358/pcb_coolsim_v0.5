## 1. 概述

### 1.1 功能定位

动态仿真计算模块基于逐时气象数据，计算全年 8760 小时 × 多年（默认 3 年）的逐时冷负荷变化，为设备选型和运行策略提供数据支撑。

**核心目标：**
- 支持两种仿真方式：气象数据驱动（方式一）和比例系数法（方式二）
- 8760h × 3 年 × 1000+ 功能区域计算 < 30 秒
- 计算精度（方式一）：与独立脚本验算偏差 < 1%
- 结果存储：TimescaleDB 超表分区存储

### 1.2 技术架构

```
动态仿真计算引擎
├── WeatherDriver           # 气象数据驱动仿真
│   ├── 数据加载器         # 加载气象数据
│   ├── 焓值计算器         # psychrolib 焓值计算
│   └── 负荷计算器         # 逐时负荷计算
├── CoefficientMethod       # 比例系数法仿真
│   ├── 系数映射表         # 统一比例系数表
│   └── 负荷计算器         # 系数应用计算
├── ResultProcessor         # 结果处理器
│   ├── 数据聚合器         # 多级聚合
│   ├── 统计计算器         # 极值统计
│   └── 持续曲线生成器     # 负荷持续曲线
└── StorageManager          # 存储管理器
    ├── TimescaleDB 写入    # 超表分区写入
    └── Redis 缓存          # 结果缓存
```

### 2.1 数据加载

#### 2.1.1 气象数据加载

```python
"""
气象数据加载器

PRD 引用：F6-001
Spec 引用：模块-06-动态仿真负荷计算.md Section 10.2
"""

from typing import List, Dict, Optional
from datetime import datetime
import numpy as np
from apps.weather.models import WeatherRecord


class WeatherDataLoader:
    """
    气象数据加载器

    加载项目关联城市的逐时气象数据，支持多年数据加载。
    """

    def __init__(self, project_id: int):
        """
        初始化气象数据加载器

        Args:
            project_id: 项目 ID
        """
        self.project_id = project_id
        self.city = self._load_project_city()

    def _load_project_city(self) -> str:
        """
        加载项目关联的城市

        Returns:
            城市名称

        Raises:
            ValueError: 项目不存在或未关联城市
        """
        from apps.projects.models import Project
        try:
            project = Project.objects.get(id=self.project_id)
            if not project.city:
                raise ValueError("项目未关联城市")
            return project.city.name
        except Project.DoesNotExist:
            raise ValueError(f"项目 {self.project_id} 不存在")

    def load_weather_data(
        self,
        years: List[int],
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None
    ) -> np.ndarray:
        """
        加载指定年份的逐时气象数据

        Args:
            years: 年份列表，如 [2023, 2024, 2025]
            start_date: 开始日期（可选）
            end_date: 结束日期（可选）

        Returns:
            结构化 numpy 数组，包含以下字段：
            - datetime: 时间戳
            - dry_bulb_temp: 干球温度 (°C)
            - wet_bulb_temp: 湿球温度 (°C)
            - relative_humidity: 相对湿度 (%)
            - atmospheric_pressure: 大气压力 (Pa)
            - wind_speed: 风速 (m/s)
            - solar_radiation: 太阳辐射 (W/m²)

        Raises:
            ValueError: 气象数据不完整
        """
        # 构建查询
        queryset = WeatherRecord.objects.filter(
            city=self.city,
            year__in=years
        ).order_by('datetime')

        # 应用日期过滤
        if start_date:
            queryset = queryset.filter(datetime__gte=start_date)
        if end_date:
            queryset = queryset.filter(datetime__lte=end_date)

        # 转换为 numpy 数组
        data = queryset.values_list(
            'datetime',
            'dry_bulb_temp',
            'wet_bulb_temp',
            'relative_humidity',
            'atmospheric_pressure',
            'wind_speed',
            'solar_radiation'
        )

        # 检查数据完整性
        expected_hours = self._calculate_expected_hours(years)
        if len(data) < expected_hours * 0.95:  # 允许 5% 缺失
            raise ValueError(
                f"气象数据不完整：期望 {expected_hours} 小时，"
                f"实际 {len(data)} 小时"
            )

        # 转换为 numpy 结构化数组
        dtypes = [
            ('datetime', 'datetime64[ns]'),
            ('dry_bulb_temp', 'f4'),
            ('wet_bulb_temp', 'f4'),
            ('relative_humidity', 'f4'),
            ('atmospheric_pressure', 'f4'),
            ('wind_speed', 'f4'),
            ('solar_radiation', 'f4')
        ]

        weather_array = np.array(data, dtype=dtypes)

        return weather_array

    def _calculate_expected_hours(self, years: List[int]) -> int:
        """
        计算期望的小时数

        Args:
            years: 年份列表

        Returns:
            期望的小时数
        """
        total_hours = 0
        for year in years:
            # 闰年 8784 小时，平年 8760 小时
            if (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0):
                total_hours += 8784
            else:
                total_hours += 8760
        return total_hours
```

#### 2.1.2 房间数据加载

```python
"""
房间数据加载器

PRD 引用：F6-002, F6-003
Spec 引用：模块-06-动态仿真负荷计算.md Section 10.2
"""

from typing import List, Dict, Optional
from decimal import Decimal
from apps.projects.models import Room
from apps.calculations.models import StaticCalculationResult


class RoomDataLoader:
    """
    房间数据加载器

    加载功能区域的静态计算结果和配置参数。
    """

    def __init__(self, project_id: int):
        """
        初始化房间数据加载器

        Args:
            project_id: 项目 ID
        """
        self.project_id = project_id

    def load_rooms(
        self,
        room_ids: Optional[List[int]] = None,
        building_ids: Optional[List[int]] = None
    ) -> Dict[int, Dict]:
        """
        加载功能区域数据

        Args:
            room_ids: 功能区域 ID 列表（可选）
            building_ids: 建筑 ID 列表（可选）

        Returns:
            字典，key 为 room_id，value 为房间配置数据
        """
        # 构建查询
        queryset = Room.objects.filter(
            floor__building__project_id=self.project_id
        ).select_related(
            'floor',
            'floor__building',
            'floor__building__project'
        )

        # 应用过滤
        if room_ids:
            queryset = queryset.filter(id__in=room_ids)
        if building_ids:
            queryset = queryset.filter(
                floor__building_id__in=building_ids
            )

        # 加载房间数据
        rooms = {}
        for room in queryset:
            # 加载静态计算结果
            static_result = self._load_static_result(room.id)

            # 加载默认配置
            default_config = self._load_default_config(room)

            rooms[room.id] = {
                'id': room.id,
                'name': room.room_name,
                'area': float(room.area),
                'ceiling_height': float(room.ceiling_height),
                'volume': float(room.area * room.ceiling_height),

                # 温湿度参数
                'indoor_temp': float(
                    room.indoor_calc_temp or
                    default_config.get('indoor_calc_temp', 26.0)
                ),
                'indoor_humidity': float(
                    room.indoor_calc_humidity or
                    default_config.get('indoor_calc_humidity', 55.0)
                ),

                # 新风参数
                'fresh_air_volume': float(
                    self._calculate_fresh_air_volume(room, default_config)
                ),

                # 静态负荷
                'terminal_load': float(
                    static_result.terminal_load if static_result else 0
                ),
                'static_fresh_air_load': float(
                    static_result.fresh_air_load if static_result else 0
                ),
                'static_total_load': float(
                    static_result.total_load if static_result else 0
                ),

                # 焓值（室内，全年恒定）
                'indoor_enthalpy': float(
                    self._calculate_indoor_enthalpy(
                        float(
                            room.indoor_calc_temp or
                            default_config.get('indoor_calc_temp', 26.0)
                        ),
                        float(
                            room.indoor_calc_humidity or
                            default_config.get('indoor_calc_humidity', 55.0)
                        ),
                        float(
                            room.floor.building.project.city.pressure_summer
                        )
                    )
                ),

                # 冷冻水配置
                'water_grade': room.water_grade or 'medium',

                # 元数据
                'floor_id': room.floor_id,
                'building_id': room.floor.building_id,
                'project_id': self.project_id
            }

        return rooms

    def _load_static_result(
        self,
        room_id: int
    ) -> Optional[StaticCalculationResult]:
        """
        加载静态计算结果

        Args:
            room_id: 功能区域 ID

        Returns:
            静态计算结果或 None
        """
        try:
            return StaticCalculationResult.objects.filter(
                room_id=room_id
            ).latest('calculated_at')
        except StaticCalculationResult.DoesNotExist:
            return None

    def _load_default_config(self, room: Room) -> Dict:
        """
        加载默认配置

        优先级：功能区域显式值 > 楼层覆盖 > 建筑覆盖 > 项目覆盖 > 系统默认

        Args:
            room: 功能区域对象

        Returns:
            默认配置字典
        """
        from apps.settings.models import DefaultConfigValue

        # 查询链
        configs = DefaultConfigValue.objects.filter(
            models.Q(level='system') |
            models.Q(level='project', project_id=self.project_id) |
            models.Q(level='building', building_id=room.floor.building_id) |
            models.Q(level='floor', floor_id=room.floor_id)
        ).order_by('-priority')

        # 合并配置
        result = {}
        for config in configs:
            if config.param_key not in result:
                result[config.param_key] = float(config.param_value)

        return result

    def _calculate_fresh_air_volume(
        self,
        room: Room,
        default_config: Dict
    ) -> float:
        """
        计算新风量

        公式：新风量 = 总排风量 + 压差渗透风量

        Args:
            room: 功能区域对象
            default_config: 默认配置

        Returns:
            新风量 (m³/h)
        """
        # 总排风量
        exhaust_volume = float(
            (room.heat_exhaust_volume or 0) +
            (room.acid_exhaust_volume or 0) +
            (room.alkali_exhaust_volume or 0) +
            (room.organic_exhaust_volume or 0) +
            (room.dust_exhaust_volume or 0)
        )

        # 压差渗透风量
        volume = float(room.area * room.ceiling_height)
        pressure_diff = float(room.pressure_diff or 0)

        # 查找压差渗透系数
        infiltration_coeff = self._get_infiltration_coefficient(
            pressure_diff,
            default_config
        )

        infiltration_volume = volume * infiltration_coeff

        return exhaust_volume + infiltration_volume

    def _get_infiltration_coefficient(
        self,
        pressure_diff: float,
        default_config: Dict
    ) -> float:
        """
        获取压差渗透风量系数

        按压差绝对值查表，离散点线性插值

        Args:
            pressure_diff: 压差 (Pa)
            default_config: 默认配置

        Returns:
            渗透系数 (次/h)
        """
        # 压差渗透风量系数表
        coeff_table = default_config.get('infiltration_coefficient_table', {
            0: 0,
            1: 1.0,
            2: 1.5,
            3: 2.0,
            5: 3.0,
            10: 3.5,
            15: 4.0
        })

        # 按绝对值查表
        abs_pressure = abs(pressure_diff)

        # 查找最近的两个点进行线性插值
        sorted_points = sorted(coeff_table.items())

        for i in range(len(sorted_points) - 1):
            p1, k1 = sorted_points[i]
            p2, k2 = sorted_points[i + 1]

            if p1 <= abs_pressure <= p2:
                # 线性插值
                ratio = (abs_pressure - p1) / (p2 - p1)
                return k1 + (k2 - k1) * ratio

        # 超过最大值，封顶
        return sorted_points[-1][1]

    def _calculate_indoor_enthalpy(
        self,
        temp: float,
        humidity: float,
        pressure: float
    ) -> float:
        """
        计算室内焓值

        使用 psychrolib 库计算

        Args:
            temp: 干球温度 (°C)
            humidity: 相对湿度 (%)
            pressure: 大气压力 (Pa)

        Returns:
            焓值 (kJ/kg)
        """
        import psychrolib

        # 设置单位系统
        psychrolib.SetUnitSystem(psychrolib.SI)

        # 计算焓值
        enthalpy = psychrolib.GetMoistAirEnthalpy(
            temp,
            humidity / 100,  # psychrolib 使用 0-1 的湿度值
            pressure
        )

        return enthalpy / 1000  # 转换为 kJ/kg
```

### 2.2 焓值计算

```python
"""
焓值计算模块

PRD 引用：F6-002
Spec 引用：模块-06-动态仿真负荷计算.md Section 10.2
计算规范：冷量仿真核心计算规范.md Section 2.3
"""

import numpy as np
from typing import Tuple
import psychrolib


class EnthalpyCalculator:
    """
    焓值计算器

    使用 psychrolib 库计算湿空气焓值，支持向量化计算。
    """

    def __init__(self):
        """初始化焓值计算器"""
        # 设置单位系统
        psychrolib.SetUnitSystem(psychrolib.SI)

    def calculate_single(
        self,
        dry_bulb_temp: float,
        relative_humidity: float,
        atmospheric_pressure: float
    ) -> float:
        """
        计算单个点的焓值

        Args:
            dry_bulb_temp: 干球温度 (°C)
            relative_humidity: 相对湿度 (0-100)
            atmospheric_pressure: 大气压力 (Pa)

        Returns:
            焓值 (kJ/kg)

        Raises:
            ValueError: 输入参数超出物理范围
        """
        # 参数校验
        if dry_bulb_temp < -50 or dry_bulb_temp > 60:
            raise ValueError(f"干球温度超出范围：{dry_bulb_temp}°C")
        if relative_humidity < 0 or relative_humidity > 100:
            raise ValueError(f"相对湿度超出范围：{relative_humidity}%")
        if atmospheric_pressure < 80000 or atmospheric_pressure > 110000:
            raise ValueError(f"大气压力超出范围：{atmospheric_pressure} Pa")

        # 计算焓值
        enthalpy = psychrolib.GetMoistAirEnthalpy(
            dry_bulb_temp,
            relative_humidity / 100,  # psychrolib 使用 0-1
            atmospheric_pressure
        )

        return enthalpy / 1000  # J/kg -> kJ/kg

    def calculate_vectorized(
        self,
        dry_bulb_temps: np.ndarray,
        relative_humidities: np.ndarray,
        atmospheric_pressures: np.ndarray
    ) -> np.ndarray:
        """
        向量化计算焓值

        Args:
            dry_bulb_temps: 干球温度数组 (°C)
            relative_humidities: 相对湿度数组 (0-100)
            atmospheric_pressures: 大气压力数组 (Pa)

        Returns:
            焓值数组 (kJ/kg)
        """
        # 转换湿度为 0-1 范围
        humidities_normalized = relative_humidities / 100

        # 向量化计算
        enthalpies = np.vectorize(
            lambda t, h, p: psychrolib.GetMoistAirEnthalpy(t, h, p)
        )(dry_bulb_temps, humidities_normalized, atmospheric_pressures)

        return enthalpies / 1000  # J/kg -> kJ/kg

    def calculate_from_wet_bulb(
        self,
        dry_bulb_temp: float,
        wet_bulb_temp: float,
        atmospheric_pressure: float
    ) -> Tuple[float, float]:
        """
        从干湿球温度计算焓值和相对湿度

        Args:
            dry_bulb_temp: 干球温度 (°C)
            wet_bulb_temp: 湿球温度 (°C)
            atmospheric_pressure: 大气压力 (Pa)

        Returns:
            (焓值 kJ/kg, 相对湿度 %)
        """
        # 计算相对湿度
        humidity_ratio = psychrolib.GetHumRatioFromTWetBulb(
            dry_bulb_temp,
            wet_bulb_temp,
            atmospheric_pressure
        )

        relative_humidity = psychrolib.GetRelHumFromHumRatio(
            dry_bulb_temp,
            humidity_ratio,
            atmospheric_pressure
        )

        # 计算焓值
        enthalpy = psychrolib.GetMoistAirEnthalpy(
            dry_bulb_temp,
            relative_humidity,
            atmospheric_pressure
        )

        return enthalpy / 1000, relative_humidity * 100

    def calculate_fresh_air_load(
        self,
        fresh_air_volume: float,
        outdoor_enthalpy: float,
        indoor_enthalpy: float,
        air_density: float = 1.2
    ) -> float:
        """
        计算新风负荷

        公式：Q_fresh_air = ρ × V × (h_outdoor - h_indoor) / 3600

        Args:
            fresh_air_volume: 新风量 (m³/h)
            outdoor_enthalpy: 室外焓值 (kJ/kg)
            indoor_enthalpy: 室内焓值 (kJ/kg)
            air_density: 空气密度 (kg/m³)，默认 1.2

        Returns:
            新风负荷 (kW)
        """
        enthalpy_diff = outdoor_enthalpy - indoor_enthalpy

        # 新风负荷可为负值（冬季室外焓低于室内）
        fresh_air_load = (
            air_density *
            fresh_air_volume *
            enthalpy_diff /
            3600
        )

        return fresh_air_load
```

### 2.3 仿真引擎

```python
"""
气象数据驱动仿真引擎

PRD 引用：F6-001 ~ F6-007
Spec 引用：模块-06-动态仿真负荷计算.md Section 10.2
"""

import numpy as np
from typing import Dict, List, Optional
from datetime import datetime
from apps.calculations.models import (
    SimulationTask,
    SimulationHourlyResult
)


class WeatherDrivenSimulator:
    """
    气象数据驱动仿真引擎

    使用历史逐时气象数据计算冷负荷变化。
    """

    def __init__(
        self,
        project_id: int,
        room_ids: Optional[List[int]] = None
    ):
        """
        初始化仿真引擎

        Args:
            project_id: 项目 ID
            room_ids: 功能区域 ID 列表（可选）
        """
        self.project_id = project_id
        self.room_ids = room_ids
        self.weather_loader = WeatherDataLoader(project_id)
        self.room_loader = RoomDataLoader(project_id)
        self.enthalpy_calculator = EnthalpyCalculator()

    def simulate(
        self,
        years: List[int],
        simulation_task_id: Optional[int] = None
    ) -> Dict[int, List[Dict]]:
        """
        执行气象数据驱动仿真

        Args:
            years: 年份列表，如 [2023, 2024, 2025]
            simulation_task_id: 仿真任务 ID（可选）

        Returns:
            字典，key 为 room_id，value 为逐时结果列表

        Raises:
            ValueError: 气象数据不完整或房间数据缺失
        """
        # 1. 加载气象数据
        weather_data = self.weather_loader.load_weather_data(years)

        # 2. 加载房间数据
        rooms = self.room_loader.load_rooms(self.room_ids)

        # 3. 初始化结果存储
        results = {room_id: [] for room_id in rooms.keys()}

        # 4. 获取项目大气压力（用于焓值计算）
        pressure = self._get_project_pressure()

        # 5. 逐小时计算
        total_hours = len(weather_data)
        for hour_idx, weather in enumerate(weather_data):
            # 5.1 计算室外焓值
            outdoor_enthalpy = self.enthalpy_calculator.calculate_single(
                float(weather['dry_bulb_temp']),
                float(weather['relative_humidity']),
                pressure
            )

            # 5.2 逐房间计算
            for room_id, room_config in rooms.items():
                # 室内焓值（全年恒定）
                indoor_enthalpy = room_config['indoor_enthalpy']

                # 5.2.1 计算新风负荷
                fresh_air_load = self.enthalpy_calculator.calculate_fresh_air_load(
                    room_config['fresh_air_volume'],
                    outdoor_enthalpy,
                    indoor_enthalpy
                )

                # 5.2.2 末端负荷（固定，来自静态计算）
                terminal_load = room_config['terminal_load']

                # 5.2.3 总负荷
                total_load = terminal_load + fresh_air_load

                # 5.2.4 负值截断（PRD: F6-006）
                # 总负荷 < 0 时截断为 0
                if total_load < 0:
                    total_load = 0
                    fresh_air_load = -terminal_load  # 新风负荷为负

                # 5.2.5 存储结果
                results[room_id].append({
                    'hour_index': hour_idx,
                    'datetime': weather['datetime'],
                    'year': weather['datetime'].year,
                    'hour_of_year': hour_idx % 8760,

                    # 气象数据
                    'outdoor_temp': float(weather['dry_bulb_temp']),
                    'outdoor_humidity': float(weather['relative_humidity']),
                    'outdoor_enthalpy': outdoor_enthalpy,

                    # 负荷结果
                    'terminal_load': terminal_load,
                    'fresh_air_load': fresh_air_load,
                    'total_load': total_load,

                    # 元数据
                    'simulation_type': 'weather_driven'
                })

            # 6. 更新进度
```

## 3. 比例系数法仿真（方式二）

### 3.1 系数映射表

```python
"""
比例系数映射表

PRD 引用：F6-008 ~ F6-011
Spec 引用：模块-06-动态仿真负荷计算.md Section 10.3
ADR 引用：ADR-0003（统一比例系数表）
"""

from typing import Dict, Tuple
import numpy as np


class CoefficientMapping:
    """
    比例系数映射表

    使用系统内置的统一映射表，维度为室外温度 × 生产负荷率。
    """

    # 默认映射表（ADR-0003）
    DEFAULT_MAPPING = {
        # (温度下限, 温度上限): {负荷率: 系数}
        (float('-inf'), 5): {
            10: 0.3, 30: 0.4, 50: 0.5, 80: 0.7, 100: 0.8
        },
        (5, 15): {
            10: 0.4, 30: 0.5, 50: 0.6, 80: 0.8, 100: 0.9
        },
        (15, 25): {
            10: 0.5, 30: 0.6, 50: 0.7, 80: 0.9, 100: 1.0
        },
        (25, 35): {
            10: 0.7, 30: 0.8, 50: 0.9, 80: 1.0, 100: 1.1
        },
        (35, float('inf')): {
            10: 0.8, 30: 0.9, 50: 1.0, 80: 1.1, 100: 1.2
        }
    }

    def __init__(self, project_id: Optional[int] = None):
        """
        初始化系数映射表

        Args:
            project_id: 项目 ID（可选，用于加载项目级覆盖）
        """
        self.project_id = project_id
        self.mapping = self._load_mapping()

    def _load_mapping(self) -> Dict:
        """
        加载比例系数映射表

        优先级：项目级覆盖 > 系统默认

        Returns:
            映射表字典
        """
        mapping = self.DEFAULT_MAPPING.copy()

        # 尝试加载项目级覆盖
        if self.project_id:
            try:
                from apps.settings.models import CoefficientMappingOverride
                overrides = CoefficientMappingOverride.objects.filter(
                    project_id=self.project_id
                )

                for override in overrides:
                    temp_range = (override.temp_min, override.temp_max)
                    if temp_range not in mapping:
                        mapping[temp_range] = {}
                    mapping[temp_range][override.production_rate] = (
                        override.coefficient
                    )
            except Exception:
                pass  # 使用默认映射表

        return mapping

    def get_coefficient(
        self,
        outdoor_temp: float,
        production_rate: float
    ) -> float:
        """
        获取比例系数

        支持双线性插值

        Args:
            outdoor_temp: 室外温度 (°C)
            production_rate: 生产负荷率 (10-100)

        Returns:
            比例系数
        """
        # 1. 找到温度区间
        temp_range = None
        for (temp_min, temp_max) in self.mapping.keys():
            if temp_min <= outdoor_temp < temp_max:
                temp_range = (temp_min, temp_max)
                break

        if not temp_range:
            # 超出范围，使用最近的区间
            if outdoor_temp < 5:
                temp_range = (float('-inf'), 5)
            else:
                temp_range = (35, float('inf'))

        # 2. 获取该区间的系数表
        rate_coefficients = self.mapping[temp_range]

        # 3. 获取标准负荷率列表
        standard_rates = sorted(rate_coefficients.keys())

        # 4. 查找最近的两个负荷率进行插值
        for i in range(len(standard_rates) - 1):
            r1 = standard_rates[i]
            r2 = standard_rates[i + 1]

            if r1 <= production_rate <= r2:
                # 线性插值
                c1 = rate_coefficients[r1]
                c2 = rate_coefficients[r2]

                ratio = (production_rate - r1) / (r2 - r1)
                coefficient = c1 + (c2 - c1) * ratio

                return coefficient

        # 超出范围
        if production_rate < standard_rates[0]:
            return rate_coefficients[standard_rates[0]]
        else:
            return rate_coefficients[standard_rates[-1]]

    def get_coefficient_vectorized(
        self,
        outdoor_temps: np.ndarray,
        production_rates: np.ndarray
    ) -> np.ndarray:
        """
        向量化获取比例系数

        Args:
            outdoor_temps: 室外温度数组 (°C)
            production_rates: 生产负荷率数组 (10-100)

        Returns:
            比例系数数组
        """
        coefficients = np.vectorize(
            lambda t, r: self.get_coefficient(t, r)
        )(outdoor_temps, production_rates)

        return coefficients
```

### 3.2 比例系数法仿真引擎

```python
"""
比例系数法仿真引擎

PRD 引用：F6-008 ~ F6-011
Spec 引用：模块-06-动态仿真负荷计算.md Section 10.3
"""

import numpy as np
from typing import Dict, List, Optional
from datetime import datetime, timedelta


class CoefficientMethodSimulator:
    """
    比例系数法仿真引擎

    使用比例系数映射表快速计算逐时负荷。
    """

    def __init__(
        self,
        project_id: int,
        room_ids: Optional[List[int]] = None
    ):
        """
        初始化比例系数法仿真引擎

        Args:
            project_id: 项目 ID
            room_ids: 功能区域 ID 列表（可选）
        """
        self.project_id = project_id
        self.room_ids = room_ids
        self.room_loader = RoomDataLoader(project_id)
        self.coefficient_mapping = CoefficientMapping(project_id)

    def simulate(
        self,
        years: List[int],
        production_schedule: Optional[Dict[int, float]] = None,
        simulation_task_id: Optional[int] = None
    ) -> Dict[int, List[Dict]]:
        """
        执行比例系数法仿真

        Args:
            years: 年份列表
            production_schedule: 生产负荷率计划，格式：
                {
                    10: 50,  # 前 10 天负荷率 50%
                    20: 80,  # 接下来 20 天负荷率 80%
                    ...
                }
            simulation_task_id: 仿真任务 ID（可选）

        Returns:
            字典，key 为 room_id，value 为逐时结果列表
        """
        # 1. 加载房间数据
        rooms = self.room_loader.load_rooms(self.room_ids)

        # 2. 生成时间序列
        time_series = self._generate_time_series(years)

        # 3. 初始化结果存储
        results = {room_id: [] for room_id in rooms.keys()}

        # 4. 逐小时计算
        total_hours = len(time_series)
        for hour_idx, dt in enumerate(time_series):
            # 4.1 计算室外温度（使用年平均温度或真实数据）
            outdoor_temp = self._get_outdoor_temp(dt)

            # 4.2 获取生产负荷率
            production_rate = self._get_production_rate(
                dt,
                production_schedule
            )

            # 4.3 获取比例系数
            coefficient = self.coefficient_mapping.get_coefficient(
                outdoor_temp,
                production_rate
            )

            # 4.4 逐房间计算
            for room_id, room_config in rooms.items():
                # 静态负荷
                static_load = room_config['static_total_load']

                # 动态负荷 = 静态负荷 × 系数
                dynamic_load = static_load * coefficient

                # 负值截断
                if dynamic_load < 0:
                    dynamic_load = 0

                # 存储结果
                results[room_id].append({
                    'hour_index': hour_idx,
                    'datetime': dt,
                    'year': dt.year,
                    'hour_of_year': hour_idx % 8760,

                    # 气象数据（近似）
                    'outdoor_temp': outdoor_temp,
                    'outdoor_humidity': None,  # 比例系数法不计算湿度
                    'outdoor_enthalpy': None,

                    # 负荷结果
                    'terminal_load': room_config['terminal_load'],
                    'fresh_air_load': dynamic_load - room_config['terminal_load'],
                    'total_load': dynamic_load,

                    # 系数信息
                    'coefficient': coefficient,
                    'production_rate': production_rate,

                    # 元数据
                    'simulation_type': 'coefficient'
                })

            # 5. 更新进度
            if simulation_task_id and hour_idx % 100 == 0:
                self._update_progress(
                    simulation_task_id,
                    hour_idx,
                    total_hours
                )

        return results

    def _generate_time_series(
        self,
        years: List[int]
    ) -> List[datetime]:
        """
        生成时间序列

        Args:
            years: 年份列表

        Returns:
            时间戳列表
        """
        time_series = []
        for year in years:
            start_date = datetime(year, 1, 1)
            end_date = datetime(year, 12, 31, 23, 0, 0)

            current = start_date
            while current <= end_date:
                time_series.append(current)
                current += timedelta(hours=1)

        return time_series

    def _get_outdoor_temp(self, dt: datetime) -> float:
        """
        获取室外温度

        简化处理：使用年平均温度或从气象数据加载

        Args:
            dt: 时间戳

        Returns:
            室外温度 (°C)
        """
        # 简化：使用年平均温度
        # 实际实现应该从气象数据加载或使用温度模型
        return 20.0  # 占位值

    def _get_production_rate(
        self,
        dt: datetime,
        production_schedule: Optional[Dict[int, float]]
    ) -> float:
        """
        获取生产负荷率

        Args:
            dt: 时间戳
            production_schedule: 生产负荷率计划

        Returns:
            生产负荷率 (10-100)
        """
        if not production_schedule:
            return 100.0  # 默认 100%

        # 根据日期确定负荷率
        day_of_year = dt.timetuple().tm_yday

        accumulated_days = 0
        for days, rate in production_schedule.items():
            accumulated_days += days
            if day_of_year <= accumulated_days:
                return rate

        return 100.0  # 默认

    def _update_progress(
        self,
        task_id: int,
        current: int,
        total: int
    ):
        """更新进度"""
        from celery import current_task
        if current_task:
            progress = int((current / total) * 100)
            current_task.update_state(
                state='PROGRESS',
                meta={
                    'current': current,
                    'total': total,
                    'progress': progress
                }
            )
```

## 4. 结果处理与聚合

### 4.1 结果聚合器

```python
"""
结果聚合器

PRD 引用：F7-001 ~ F7-016
Spec 引用：模块-07-8760可视化与基础验证.md Section 11.2
"""

import numpy as np
from typing import Dict, List, Tuple
from datetime import datetime
from collections import defaultdict


class ResultAggregator:
    """
    结果聚合器

    将逐时结果聚合为多种粒度：日、月、年。
    """

    def __init__(self, hourly_results: Dict[int, List[Dict]]):
        """
        初始化结果聚合器

        Args:
            hourly_results: 逐时结果，key 为 room_id
        """
        self.hourly_results = hourly_results

    def aggregate_daily(self) -> Dict[int, Dict]:
        """
        按日聚合

        Returns:
            字典，key 为 room_id，value 为每日统计数据
        """
        daily_stats = {}

        for room_id, data in self.hourly_results.items():
            daily_stats[room_id] = self._aggregate_by_period(
                data,
                lambda dt: dt.date()
            )

        return daily_stats

    def aggregate_monthly(self) -> Dict[int, Dict]:
        """
        按月聚合

        Returns:
            字典，key 为 room_id，value 为每月统计数据
        """
        monthly_stats = {}

        for room_id, data in self.hourly_results.items():
            monthly_stats[room_id] = self._aggregate_by_period(
                data,
                lambda dt: (dt.year, dt.month)
            )

        return monthly_stats

    def aggregate_yearly(self) -> Dict[int, Dict]:
        """
        按年聚合

        Returns:
            字典，key 为 room_id，value 为每年统计数据
        """
        yearly_stats = {}

        for room_id, data in self.hourly_results.items():
            yearly_stats[room_id] = self._aggregate_by_period(
                data,
                lambda dt: dt.year
            )

        return yearly_stats

    def _aggregate_by_period(
        self,
        data: List[Dict],
        period_key_func
    ) -> Dict:
        """
        按时间段聚合

        Args:
            data: 逐时数据
            period_key_func: 时间段键函数

        Returns:
            聚合统计结果
        """
        period_groups = defaultdict(list)

        for entry in data:
            period_key = period_key_func(entry['datetime'])
            period_groups[period_key].append(entry['total_load'])

        stats = {}
        for period_key, loads in period_groups.items():
            loads_array = np.array(loads)
            stats[period_key] = {
                'max_load': float(np.max(loads_array)),
                'min_load': float(np.min(loads_array[loads_array > 0])),  # 有意义最小值
                'avg_load': float(np.mean(loads_array)),
                'total_energy': float(np.sum(loads_array)),  # kWh
                'peak_hour': int(np.argmax(loads_array)),
                'load_factor': float(np.mean(loads_array) / np.max(loads_array))
            }

        return stats

    def calculate_peak_analysis(self) -> Dict[int, Dict]:
        """
        计算极值分析

        Returns:
            字典，key 为 room_id，value 为极值统计
        """
        peak_stats = {}

        for room_id, data in self.hourly_results.items():
            loads = np.array([entry['total_load'] for entry in data])
            datetimes = [entry['datetime'] for entry in data]

            # 最大负荷
            max_idx = np.argmax(loads)
            max_load = float(loads[max_idx])
            max_datetime = datetimes[max_idx]

            # 最小有意义负荷（> 0）
            positive_loads = loads[loads > 0]
            if len(positive_loads) > 0:
                min_idx = np.argmin(positive_loads)
                min_load = float(positive_loads[min_idx])
                # 找到对应的 datetime
                min_datetime_idx = np.where(loads == min_load)[0][0]
                min_datetime = datetimes[min_datetime_idx]
            else:
                min_load = 0
                min_datetime = None

            # 平均负荷
            avg_load = float(np.mean(loads))

            # 负荷率
            load_factor = avg_load / max_load if max_load > 0 else 0

            peak_stats[room_id] = {
                'max_load': max_load,
                'max_datetime': max_datetime,
                'min_load': min_load,
                'min_datetime': min_datetime,
                'avg_load': avg_load,
                'load_factor': load_factor,
                'total_energy': float(np.sum(loads))
            }

        return peak_stats

    def calculate_duration_curve(
        self,
        room_id: int
    ) -> List[Tuple[float, float]]:
        """
        计算负荷持续曲线

        Args:
            room_id: 功能区域 ID

        Returns:
            列表，每项为 (负荷率, 累积小时数)
        """
        data = self.hourly_results.get(room_id, [])
        if not data:
            return []

        loads = np.array([entry['total_load'] for entry in data])
        max_load = np.max(loads)

        if max_load == 0:
            return []

        # 计算负荷率
        load_ratios = loads / max_load

        # 排序
        sorted_ratios = np.sort(load_ratios)[::-1]

        # 计算累积小时数
        hours = np.arange(1, len(sorted_ratios) + 1)

        # 生成持续曲线数据点
        duration_curve = []
        for i in range(len(sorted_ratios)):
            duration_curve.append((
                float(sorted_ratios[i]),
                int(hours[i])
            ))

        return duration_curve

    def calculate_percentiles(self) -> Dict[int, Dict]:
        """
        计算百分位数

        Returns:
            字典，key 为 room_id，value 为百分位统计
        """
        percentiles = {}

        for room_id, data in self.hourly_results.items():
            loads = np.array([entry['total_load'] for entry in data])

            percentiles[room_id] = {
                'p10': float(np.percentile(loads, 90)),  # 前 10% 峰值
                'p25': float(np.percentile(loads, 75)),
                'p50': float(np.percentile(loads, 50)),  # 中位数
                'p75': float(np.percentile(loads, 25)),
                'p90': float(np.percentile(loads, 10)),
                'p95': float(np.percentile(loads, 5)),
                'p99': float(np.percentile(loads, 1))
            }

        return percentiles
```

### 4.2 LTTB 降采样算法

```python
"""
LTTB 降采样算法

PRD 引用：F7-012 ~ F7-016
Spec 引用：模块-07-8760可视化与基础验证.md Section 11.3
"""

import numpy as np
from typing import List, Tuple


class LTTBDownsampler:
    """
    LTTB (Largest Triangle Three Buckets) 降采样算法

    保留数据的主要形状特征，同时减少数据点数量。
    """

    def downsample(
        self,
        data: List[Tuple[float, float]],
        target_points: int
    ) -> List[Tuple[float, float]]:
        """
        执行 LTTB 降采样

        Args:
            data: 原始数据点列表，每项为 (x, y)
            target_points: 目标数据点数量

        Returns:
            降采样后的数据点列表
        """
        if len(data) <= target_points:
            return data

        # 确保首尾点保留
        sampled = [data[0]]

        # 计算每个 bucket 的大小
        bucket_size = (len(data) - 2) / (target_points - 2)

        for i in range(1, target_points - 1):
            # 当前 bucket 的范围
            bucket_start = int((i - 1) * bucket_size) + 1
            bucket_end = min(int(i * bucket_size) + 1, len(data) - 1)

            # 下一个 bucket 的平均点
            next_bucket_start = int(i * bucket_size) + 1
            next_bucket_end = min(
                int((i + 1) * bucket_size) + 1,
                len(data) - 1
            )

            # 计算下一个 bucket 的平均坐标
            avg_x = 0
            avg_y = 0
            for j in range(next_bucket_start, next_bucket_end):
                avg_x += data[j][0]
                avg_y += data[j][1]
            avg_x /= (next_bucket_end - next_bucket_start)
            avg_y /= (next_bucket_end - next_bucket_start)

            # 在当前 bucket 中选择最大三角形面积的点
            max_area = -1
            selected_idx = bucket_start

            prev_point = sampled[-1]

            for j in range(bucket_start, bucket_end):
                # 计算三角形面积
                area = abs(
                    (prev_point[0] - avg_x) * (data[j][1] - prev_point[1]) -
                    (prev_point[0] - data[j][0]) * (avg_y - prev_point[1])
                ) * 0.5

                if area > max_area:
                    max_area = area
                    selected_idx = j

            sampled.append(data[selected_idx])

        # 保留末尾点
        sampled.append(data[-1])

        return sampled

    def downsample_numpy(
        self,
        x: np.ndarray,
        y: np.ndarray,
        target_points: int
    ) -> Tuple[np.ndarray, np.ndarray]:
        """
        NumPy 向量化降采样

        Args:
            x: x 坐标数组
            y: y 坐标数组
            target_points: 目标数据点数量

        Returns:
            (降采样后的 x 数组, 降采样后的 y 数组)
        """
        if len(x) <= target_points:
            return x, y

        # 转换为列表格式
        data = list(zip(x, y))

        # 执行降采样
        sampled = self.downsample(data, target_points)

        # 分离 x 和 y
        sampled_x = np.array([p[0] for p in sampled])
        sampled_y = np.array([p[1] for p in sampled])

        return sampled_x, sampled_y
```

```python
"""

    simulation_type: str,  # 'weather_driven' or 'coefficient'
    config: Dict
) -> Dict:
    """
    执行动态仿真任务

    Args:
        self: Celery 任务实例
        project_id: 项目 ID
        simulation_type: 仿真类型
        config: 配置参数

    Returns:
        任务结果
    """
    task_id = self.request.id

    try:
        # 创建仿真任务记录
        simulation_task = SimulationTask.objects.create(
            project_id=project_id,
            task_id=task_id,
            simulation_type=simulation_type,
            status='processing',
            config=config
        )

        # 根据仿真类型执行
        if simulation_type == 'weather_driven':
            simulator = WeatherDrivenSimulator(
                project_id,
                config.get('room_ids')
            )
            results = simulator.simulate(
                config['years'],
                simulation_task.id
            )
        elif simulation_type == 'coefficient':
            simulator = CoefficientMethodSimulator(
                project_id,
                config.get('room_ids')
            )
            results = simulator.simulate(
                config['years'],
                config.get('production_schedule'),
                simulation_task.id
            )
        else:
            raise ValueError(f"不支持的仿真类型：{simulation_type}")

        # 存储结果
        simulator.store_results(results)

        # 聚合结果
        aggregator = ResultAggregator(results)
        aggregated = {
            'daily': aggregator.aggregate_daily(),
            'monthly': aggregator.aggregate_monthly(),
            'yearly': aggregator.aggregate_yearly(),
            'peak': aggregator.calculate_peak_analysis(),
            'percentiles': aggregator.calculate_percentiles()
        }

        # 更新任务状态
        simulation_task.status = 'completed'
        simulation_task.completed_at = datetime.now()
        simulation_task.save()

        return {
            'success': True,
            'project_id': project_id,
            'simulation_type': simulation_type,
            'total_hours': sum(len(v) for v in results.values()),
            'total_rooms': len(results),
            'aggregated': aggregated
        }

    except Exception as e:
        # 更新任务状态为失败
        if 'simulation_task' in locals():
            simulation_task.status = 'failed'
            simulation_task.error_message = str(e)
            simulation_task.save()

        # 重新抛出异常
        raise
```

```python
    """计算异常"""
```

### 7.2 异常处理策略

```python
"""
异常处理策略
"""

def handle_simulation_error(error: Exception, context: Dict):
    """
    处理仿真异常

    Args:
        error: 异常对象
        context: 上下文信息
    """
    if isinstance(error, WeatherDataError):
        # 气象数据缺失
        logger.error(f"气象数据错误：{error}")
        raise SimulationError(
            f"气象数据不完整，请先获取 {context['years']} 年的气象数据"
        )

    elif isinstance(error, RoomDataError):
        # 房间数据错误
        logger.error(f"房间数据错误：{error}")
        raise SimulationError(
            f"功能区域 {context['room_id']} 参数缺失"
        )

    elif isinstance(error, CalculationError):
        # 计算错误
        logger.error(f"计算错误：{error}")
        # 标记该房间为计算失败，继续处理其他房间
        return None

    elif isinstance(error, StorageError):
        # 存储错误
        logger.error(f"存储错误：{error}")
        # 重试 3 次
        for attempt in range(3):
            try:
                # 重新存储
                pass
            except StorageError:
                if attempt == 2:
                    raise

    else:
        # 未知错误
        logger.error(f"未知错误：{error}")
        raise
```
