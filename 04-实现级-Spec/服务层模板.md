## 1. 概述

服务层封装业务逻辑，承接控制器层（ViewSet）与数据访问层（ORM）。模板统一约定：**输入校验 -> 事务边界 -> 领域计算 -> 统一返回结构**。以「房间静态计算服务」为例，公式严格来自《静态冷量计算实现.md》与 PRD F4-001~F4-014。

---

## 2. 统一返回结构

禁止在 service 内直接返回 Django Response。统一使用 `ServiceResult` 数据类，控制器层再包装为 API 信封（见控制器层模板）。

```python
# services/result.py
from dataclasses import dataclass
from typing import Any


@dataclass
class ServiceResult:
    success: bool
    data: Any = None
    error_code: str = ""      # 对应 API总览 错误码（如 CALCULATION_ERROR）
    message: str = ""

    @classmethod
    def ok(cls, data=None):
        return cls(success=True, data=data)

    @classmethod
    def fail(cls, error_code, message):
        return cls(success=False, error_code=error_code, message=message)
```

## 3. 服务类模板（输入校验 + 事务 + 返回）

```python
# services/base.py
from django.db import transaction
from .result import ServiceResult


class BaseService:
    # 子类领域方法内部使用 transaction.atomic 包裹写操作
    def _atomic(self, func, *args, **kwargs):
        with transaction.atomic():
            return func(*args, **kwargs)
```

## 4. 示例：房间静态计算服务（RoomStaticLoadService）

对应 PRD F4-001~F4-006（末端负荷）、F4-013（新风冷负荷，rho 取 DefaultConfig）、F4-014（负值截断）。计算维度 `room_id x season_id`，结果写入 room_load_result / room_air_volume_result / room_load_summary。

```python
# services/static_load.py
from decimal import Decimal
from django.db import transaction
from .result import ServiceResult
from ..models.room import Room
from ..models.load_param import RoomLoadParameter
from ..models.load_result import RoomLoadResult
from ..models.air_volume_result import RoomAirVolumeResult
from ..models.load_summary import RoomLoadSummary
from ..models.config import DefaultConfig
from ..models.water_temp import WaterTemperatureConfig


class RoomStaticLoadService:
    """单房间静态冷量计算（F4-001~F4-014）。"""

    def calculate(self, room_id: int, season_id: int) -> ServiceResult:
        room = Room.objects.filter(id=room_id).first()
        if not room:
            return ServiceResult.fail("RESOURCE_NOT_FOUND", f"房间 {room_id} 不存在")

        param = RoomLoadParameter.objects.filter(
            room_id=room_id, season_id=season_id
        ).first()
        if not param:
            return ServiceResult.fail("VALIDATION_ERROR", "缺少负荷计算参数，无法计算")

        try:
            with transaction.atomic():
                self._calc_terminal(room, param)
                self._calc_air_volume(room, param)
                self._calc_fresh_air_and_summary(room, season_id)
            return ServiceResult.ok({"room_id": room_id, "season_id": season_id})
        except ValueError as exc:
            return ServiceResult.fail("CALCULATION_ERROR", str(exc))

    # Step 2：末端负荷（F4-001~F4-006）
    def _calc_terminal(self, room, param) -> RoomLoadResult:
        area = Decimal(str(room.area))
        civil = param.civil_load_index * area / Decimal(1000)          # F4-001
        lighting = param.lighting_load_index * area / Decimal(1000)    # F4-002
        personnel = (
            (param.personnel_load_index or 0) * (param.personnel_count or 0)
            / Decimal(1000)
        ) if param.personnel_load_index else Decimal(0)               # F4-003
        electric = (
            (param.electric_equipment_power or 0)
            * (param.electric_equipment_coefficient or 0)
        )                                                              # F4-004
        heated = (
            (param.heated_equipment_power_with_exhaust or 0)
            * (param.heated_equipment_coefficient_with_exhaust or 0)
            + (param.heated_equipment_power_without_exhaust or 0)
            * (param.heated_equipment_coefficient_without_exhaust or 0)
        )                                                              # F4-005
        terminal = civil + lighting + personnel + electric + heated    # F4-006

        obj, _ = RoomLoadResult.objects.update_or_create(
            room_id=room.id, season_id=param.season_id,
            defaults={
                "civil_load": civil, "lighting_load": lighting,
                "personnel_load": personnel, "electric_equipment_load": electric,
                "heated_equipment_load": heated, "terminal_load": terminal,
            },
        )
        return obj

    # Step 4：新风冷负荷（F4-013），rho 取自 DefaultConfig（F2-043）
    def _calc_fresh_air_and_summary(self, room, season_id):
        air = RoomAirVolumeResult.objects.get(room_id=room.id, season_id=season_id)
        rho = Decimal(str(DefaultConfig.get("air_density", "1.2")))  # kg/m3
        enthalpy_diff = (air.outdoor_enthalpy_calc or 0) - (air.indoor_enthalpy_calc or 0)
        # 单水温模式：新风冷负荷全部归入该冷冻水分类（F4-021）
        project_id = room.floor.building.project_id
        for wt in WaterTemperatureConfig.objects.filter(project_id=project_id):
            fresh_air_load = (
                rho * Decimal(str(air.fresh_air_volume or 0))
                * Decimal(str(enthalpy_diff)) / Decimal(3600)
            )                                                          # F4-013
            terminal = RoomLoadResult.objects.get(
                room_id=room.id, season_id=season_id
            ).terminal_load
            total = terminal + fresh_air_load
            if total < 0:
                total = Decimal(0)                                    # F4-014 截断
            RoomLoadSummary.objects.update_or_create(
                room_id=room.id, season_id=season_id, water_temp_config_id=wt.id,
                defaults={
                    "fresh_air_load": fresh_air_load,
                    "terminal_load": terminal,
                    "total_load": total,
                },
            )
```

**调用方（控制器）**：

```python
result = RoomStaticLoadService().calculate(room_id, season_id)
if not result.success:
    return error_response(result.error_code, result.message, status=422)
return Response({"success": True, "data": result.data})
```
