## 1. 概述

服务层封装业务逻辑，承接控制器层（ViewSet）与数据访问层（ORM）。模板统一约定：**输入校验 -> 事务边界 -> 领域计算 -> 统一返回结构**。以「房间静态计算服务」为例，公式严格来自《静态冷量计算实现.md》与 PRD F4-001~F4-014。

---

## 2. 统一返回结构

禁止在 service 内直接返回 Django Response。统一使用 `ServiceResult` 数据类，控制器层再包装为 API 信封（见控制器层模板）。

```python
# services/result.py
from dataclasses import dataclass
from typing import Any


@dataclass
class ServiceResult:
    success: bool
    data: Any = None
    error_code: str = ""      # 对应 API总览 错误码（如 CALCULATION_ERROR）
    message: str = ""

    @classmethod
    def ok(cls, data=None):
        return cls(success=True, data=data)

    @classmethod
    def fail(cls, error_code, message):
        return cls(success=False, error_code=error_code, message=message)
```

---

## 3. 服务类模板（输入校验 + 事务 + 返回）

```python
# services/base.py
from django.db import transaction
from .result import ServiceResult


class BaseService:
    # 子类领域方法内部使用 transaction.atomic 包裹写操作
    def _atomic(self, func, *args, **kwargs):
        with transaction.atomic():
            return func(*args, **kwargs)
```

## 4. 示例：房间静态计算服务（RoomStaticLoadService）

对应 PRD F4-001~F4-006（末端负荷 5 项）、F4-007~F4-009（风量 8 点查表 + 简单加法）、F4-010~F4-013（新风冷负荷，ρ 取 DefaultConfig）、F4-014（负值截断）、F4-020~F4-023（按水温分类 ADR-0001）。计算维度 `room_id`（无 season），结果写入 RoomCalcResult / RoomLoadByWaterTemp。

```python
# services/static_load.py
from decimal import Decimal
from django.db import transaction
from .result import ServiceResult
from ..models.room import Room
from ..models.calc_result import RoomCalcResult
from ..models.load_by_water_temp import RoomLoadByWaterTemp
from ..models.config import DefaultConfig
from .engine import (
    calculate_enthalpy, lookup_infiltration_coeff,
)


# PRD §8.3 压差渗透系数 8 点表（唯一权威表）
INFILTRATION_TABLE = [(0, 0.0), (1, 1.0), (2, 1.5), (3, 2.0),
                      (5, 3.0), (10, 3.5), (15, 4.0)]


class RoomStaticLoadService:
    """单房间静态冷量计算（F4-001~F4-023）。"""

    def calculate(self, room_id: int) -> ServiceResult:
        room = Room.objects.select_related(
            "floor__building__project__city"
        ).filter(id=room_id).first()
        if not room:
            return ServiceResult.fail("RESOURCE_NOT_FOUND", f"房间 {room_id} 不存在")

        try:
            with transaction.atomic():
                self._calc_terminal(room)          # Step 2（末端桥梁，静态动态共用）
                self._calc_air_volume(room)        # Step 3
                self._calc_fresh_air_and_summary(room)  # Step 1/4/5 + 按水温分类
            return ServiceResult.ok({"room_id": room_id})
        except ValueError as exc:
            return ServiceResult.fail("CALCULATION_ERROR", str(exc))

    # Step 2：末端负荷 5 项（F4-001~F4-006）；字段全部直接读 Room 单表
    def _calc_terminal(self, room) -> Decimal:
        r = room
        area = Decimal(str(r.area or 0))
        civil = Decimal(str(r.civil_load_index or 0)) * area / Decimal(1000)       # F4-001
        lighting = Decimal(str(r.lighting_load_index or 0)) * area / Decimal(1000) # F4-002
        personnel = (
            Decimal(str(r.personnel_load_index or 0))
            * Decimal(str(r.personnel_count or 0)) / Decimal(1000)
        )                                                                           # F4-003
        electric = (
            Decimal(str(r.electric_equipment_power or 0))
            * Decimal(str(r.electric_equipment_coefficient or 0))
        )                                                                           # F4-004 kW 不乘1000
        heated = (
            Decimal(str(r.heated_equipment_power_with_exhaust or 0))
            * Decimal(str(r.heated_equipment_coefficient_with_exhaust or 0))
            + Decimal(str(r.heated_equipment_power_without_exhaust or 0))
            * Decimal(str(r.heated_equipment_coefficient_without_exhaust or 0))
        )                                                                           # F4-005 kW 不乘1000
        terminal = civil + lighting + personnel + electric + heated                # F4-006

        RoomCalcResult.objects.update_or_create(
            room_id=room.id,
            defaults={
                "civil_load": civil, "lighting_load": lighting,
                "personnel_load": personnel, "electric_equipment_load": electric,
                "heated_equipment_load": heated, "terminal_load": terminal,
            },
        )
        return terminal

    # Step 3：风量（F4-007~F4-009，8 点查表 + 简单加法，无 MAX/卫生/净化）
    def _calc_air_volume(self, room):
        r = room
        total_exhaust = sum(
            Decimal(str(getattr(r, f, 0) or 0)) for f in (
                "heat_exhaust_volume", "acid_exhaust_volume",
                "alkali_exhaust_volume", "organic_exhaust_volume", "dust_exhaust_volume",
            )
        )                                                                          # F4-007
        k = lookup_infiltration_coeff(r.pressure_diff or 0, INFILTRATION_TABLE)    # F4-008 八点插值
        infiltration = Decimal(str(r.volume or 0)) * Decimal(str(k))
        fresh = total_exhaust + infiltration                                        # F4-009 简单加法

        RoomCalcResult.objects.filter(room_id=room.id).update(
            total_exhaust_volume=total_exhaust,
            infiltration_air_volume=infiltration,
            fresh_air_volume=fresh,
        )
        return fresh

    # Step 1 + Step 4 + Step 5 + 按水温分类归档（F4-013 / F4-014 / F4-020~F4-023）
    def _calc_fresh_air_and_summary(self, room):
        city = room.floor.building.project.city
        rho = Decimal(str(DefaultConfig.get("air_density", 1.2)))   # F2-043，统一 1.2
        result = RoomCalcResult.objects.get(room_id=room.id)

        # Step 1 焓值（室内默认 26°C / 55%）
        h_out = calculate_enthalpy(
            city.temp_dry_ac_summer, city.temp_wet_ac_summer,
            city.pressure_summer, input_type='wb')
        h_in = calculate_enthalpy(
            room.indoor_calc_temp or 26.0, room.indoor_calc_humidity or 55.0,
            city.pressure_summer, input_type='rh')                   # 默认 55%，不是 60%
        result.outdoor_enthalpy_calc = h_out
        result.indoor_enthalpy_calc = h_in
        result.enthalpy_diff = h_out - h_in
        result.save()

        # Step 4 新风冷负荷（唯一公式：定义A，ρ=1.2，可为负）
        fresh_air_load = rho * Decimal(str(result.fresh_air_volume)) \
                         * Decimal(str(result.enthalpy_diff)) / Decimal(3600)   # F4-013
        terminal = result.terminal_load

        # 按水温分类归档（ADR-0001）：末端归 Room.terminal_water_temp_config_id，
        # 新风按 fresh_air 配置归档。绝不能把同一 terminal_load 复制到所有水温档。
        self._classify_by_water_temp(room, terminal, fresh_air_load)

    # ADR-0001 归档：末端按 terminal_water_temp_config_id（默认中温 12/17°C），
    # 新风按 fresh_air 配置（单水温全归该档 / 两阶段按比例，h_intermediate 预留）
    def _classify_by_water_temp(self, room, terminal_load, fresh_air_load):
        # 末端归档：默认中温，可逐区指定
        terminal_wt_id = room.terminal_water_temp_config_id or self._default_mid_temp_id(room)

        if room.fresh_air_water_mode == "two_stage":
            # 两阶段：按配置比例分配（F4-022），h_intermediate 预留接口
            ratio = Decimal(str(room.fresh_air_split_ratio or 0.5))
            fresh_wt1 = fresh_air_load * ratio
            fresh_wt2 = fresh_air_load * (Decimal(1) - ratio)
            self._upsert(room.id, terminal_wt_id,
                         terminal_load if terminal_wt_id == room.fresh_air_water_temp_config_id else Decimal(0),
                         fresh_wt1 if room.fresh_air_water_temp_config_id else Decimal(0))
            self._upsert(room.id, room.fresh_air_water_temp_config_id_2,
                         Decimal(0), fresh_wt2)
        else:
            # 单水温模式：新风全归 fresh_air_water_temp_config_id（F4-021）
            fresh_wt_id = room.fresh_air_water_temp_config_id or terminal_wt_id
            self._upsert(room.id, terminal_wt_id, terminal_load,
                         fresh_air_load if fresh_wt_id == terminal_wt_id else Decimal(0))
            if fresh_wt_id != terminal_wt_id:
                self._upsert(room.id, fresh_wt_id, Decimal(0), fresh_air_load)

    def _upsert(self, room_id, water_temp_config_id, terminal_load, fresh_air_load):
        total = terminal_load + fresh_air_load
        if total < 0:
            total = Decimal(0)                                       # F4-014 截断为 0
        RoomLoadByWaterTemp.objects.update_or_create(
            room_id=room_id, water_temp_config_id=water_temp_config_id,
            defaults={
                "terminal_load": terminal_load,
                "fresh_air_load": fresh_air_load,
                "total_load": total,
            },
        )

    def _default_mid_temp_id(self, room):
        # 取项目中温 12/17°C 配置 id 作为末端默认归档
        from ..models.water_temp import WaterTempConfig
        wt = WaterTempConfig.objects.filter(
            project_id=room.floor.building.project_id, name__icontains="中温"
        ).first()
        return wt.id if wt else None
```

**调用方（控制器）**：

```python
result = RoomStaticLoadService().calculate(room_id)
if not result.success:
    return error_response(result.error_code, result.message, status=422)
return Response({"success": True, "data": result.data})
```
