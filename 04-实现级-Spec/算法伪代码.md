## 2. 公共工具

```python
import psychrolib
psychrolib.SetUnitSystem(psychrolib.SI)
RHO = 1.2            # 空气密度 kg/m³（待确认 1.2 / 1.12）
SEC_PER_HOUR = 3600

def calculate_enthalpy(temp_dry, humidity_or_wet_bulb, pressure_hpa, input_type='rh'):
    p_pa = pressure_hpa * 100
    if input_type == 'rh':
        return psychrolib.GetMoistAirEnthalpy(
            temp_dry,
            psychrolib.GetHumRatioFromRelHum(temp_dry, humidity_or_wet_bulb, p_pa),
            p_pa)
    else:  # 'wb'
        hr = psychrolib.GetHumRatioFromTWetBulb(temp_dry, humidity_or_wet_bulb, p_pa)
        return psychrolib.GetMoistAirEnthalpy(temp_dry, hr, p_pa)

class CalcStep(ABC):
    @abstractmethod
    def get_inputs(self) -> list[str]: ...
    @abstractmethod
    def get_outputs(self) -> list[str]: ...
    @abstractmethod
    def execute(self, ctx: CalcContext) -> dict: ...
```

## 3. 静态冷量计算伪代码（F4-001）

```python
def static_calc_pipeline(room_id, season_id, water_temp_config_id):
    ctx = CalcContext.load(room_id, season_id)

    # Step 1 焓值
    h_out = calculate_enthalpy(ctx.city.temp_dry_ac_summer, ctx.city.temp_wet_ac_summer,
                               ctx.city.pressure_summer, 'wb')
    h_in  = calculate_enthalpy(ctx.room.indoor_calc_temp, ctx.room.indoor_calc_humidity,
                               ctx.city.pressure_summer, 'rh')
    ctx.set('room_air_volume_result.outdoor_enthalpy_calc', h_out)
    ctx.set('room_air_volume_result.indoor_enthalpy_calc', h_in)
    ctx.set('room_air_volume_result.enthalpy_diff', h_out - h_in)

    # Step 2 末端负荷
    p = ctx.room_load_parameter
    civil    = p.civil_load_index * ctx.room.area / 1000
    lighting = p.lighting_load_index * ctx.room.area / 1000
    personnel = (p.personnel_load_index or 0) * (p.personnel_count or 0) / 1000
    electric  = (p.electric_equipment_power or 0) * (p.electric_equipment_coefficient or 0)
    heated   = (p.heated_equipment_power_with_exhaust or 0) * (p.heated_equipment_coefficient_with_exhaust or 0) \
             + (p.heated_equipment_power_without_exhaust or 0) * (p.heated_equipment_coefficient_without_exhaust or 0)
    terminal = civil + lighting + personnel + electric + heated
    ctx.set('room_load_result.terminal_load', terminal)

    # Step 3 风量
    avp = ctx.room_air_volume_parameter
    total_exhaust = sum(x or 0 for x in [avp.heat_exhaust_volume, avp.acid_exhaust_volume,
                       avp.alkali_exhaust_volume, avp.organic_exhaust_volume, avp.dust_exhaust_volume])
    infiltration = ctx.room.volume * (ctx.pressure_diff / 5)   # k 映射待确认（静态 §3.3）
    fresh  = total_exhaust + infiltration                       # 定义 A（待确认 MAX）
    supply = max(terminal * SEC_PER_HOUR / (RHO * 1.006 * avp.supply_air_temp_diff),
                 ctx.room.volume * avp.air_change_rate)
    return_air = supply - fresh
    ctx.set('room_air_volume_result.fresh_air_volume', fresh)

    # Step 4 新风冷负荷（定义 A）
    fresh_load = RHO * fresh * (h_out - h_in) / SEC_PER_HOUR
    ctx.set('room_load_summary.fresh_air_load', fresh_load)

    # Step 5 房间总负荷
    total = fresh_load + terminal
    ctx.set('room_load_summary.total_load', total)

    # Step 6 多级聚合 → load_summary
    aggregate_to_load_summary(ctx)
    return ctx.persist()
```

```python
def simulate_weather_driven(project_id, years, room_ids=None):
    batch = create_calc_batch(project_id, 'weather_driven', years)
    weather = load_weather_series(project_id, years)          # WeatherRecord 序列
    rooms = load_rooms_static(project_id, room_ids)           # terminal_load, fresh_air_volume, h_in
    T = len(weather)
    buf = []
    for t in range(T):
        h_out = vectorized_enthalpy(weather[t])              # F6-002
        for r in rooms:
            q_fresh = RHO * r.fresh_air_volume * (h_out - r.h_indoor) / SEC_PER_HOUR  # F6-004
            q_total = r.terminal_load + q_fresh                                       # F6-005
            if q_total < 0: q_total = 0.0                                                # F6-006
            buf.append((batch.id, r.id, None, None, 'room', r.water_temp_config_id,
                        'weather_driven', weather[t].datetime,
                        weather[t].temp_dry, h_out, q_fresh, r.terminal_load, q_total))
        if len(buf) >= 2000:
            flush_dynamic_load_hourly(buf); buf.clear()
    flush_dynamic_load_hourly(buf)
    aggregate_dynamic(batch.id)                              # 见 §5
    return batch.id
```

### 4.2 比例系数法主循环

```python
def simulate_coefficient(project_id, years, room_ids=None):
    batch = create_calc_batch(project_id, 'coefficient', years)
    weather = load_weather_series(project_id, years)
    rooms = load_rooms_static(project_id, room_ids)
    coeff = load_coefficient_table(project_id)               # ADR-0003
    buf = []
    for t in range(len(weather)):
        k = coeff.lookup(weather[t].temp_dry, rate=1.0)      # F6-009
        for r in rooms:
            q = r.q_static * k                                # F6-010
            if q < 0: q = 0.0
            buf.append((batch.id, r.id, None, None, 'room', r.water_temp_config_id,
                        'coefficient', weather[t].datetime, weather[t].temp_dry,
                        None, None, r.q_static, q))
        if len(buf) >= 2000:
            flush_dynamic_load_hourly(buf); buf.clear()
    flush_dynamic_load_hourly(buf)
    return batch.id
```

### 5.2 按水温 / 区域 / 时段聚合

```python
def aggregate_dynamic(batch_id):
    # 房间级按水温分类（对应 room_load_summary 语义）
    INSERT INTO dynamic_load_hourly (... scope='room' ...)
    SELECT project_id, calc_batch_id, room_id, 'room', water_temp_config_id,
           simulation_type, ts, SUM(fresh_air_load), SUM(terminal_load), SUM(total_load)
    FROM dynamic_load_hourly
    WHERE calc_batch_id=%s AND scope='room'
    GROUP BY room_id, water_temp_config_id, ts;

    # 建筑级（楼层聚合 + extra_load）
    INSERT ... scope='building'
    SELECT ..., building_id, ts, SUM(total_load) + extra_load(building_id)
    FROM dynamic_load_hourly JOIN room ON ... JOIN extra_load ON ...
    WHERE scope='room' GROUP BY building_id, ts;

    # 项目级（所有建筑 + extra_load(building_id 空)）
    INSERT ... scope='project'
    SELECT ..., NULL, ts, SUM(total_load) + extra_load(NULL)
    GROUP BY ts;
```
