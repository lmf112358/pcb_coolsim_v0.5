## 2. 公共工具

```python
import psychrolib
psychrolib.SetUnitSystem(psychrolib.SI)

# 空气密度统一从 DefaultConfig.air_density 读取（默认 1.2 kg/m³），删除 1.2 vs 1.12 歧义（PRD §15.3）
RHO = float(DefaultConfig.get('air_density', 1.2))
SEC_PER_HOUR = 3600

def calculate_enthalpy(temp_dry, humidity_or_wet_bulb, pressure_hpa, input_type='rh'):
    p_pa = pressure_hpa * 100
    if input_type == 'rh':
        return psychrolib.GetMoistAirEnthalpy(
            temp_dry,
            psychrolib.GetHumRatioFromRelHum(temp_dry, humidity_or_wet_bulb, p_pa),
            p_pa)
    else:  # 'wb'
        hr = psychrolib.GetHumRatioFromTWetBulb(temp_dry, humidity_or_wet_bulb, p_pa)
        return psychrolib.GetMoistAirEnthalpy(temp_dry, hr, p_pa)

def lookup_infiltration_coeff(abs_pressure_diff):
    """压差渗透系数 k：PRD §8.3 / §15.2 唯一权威表，8 点线性插值。
    取 |pressure_diff| 查表；正/负压按绝对值对称；>|±15| 封顶 4.0。
    绝不能用 k=pressure_diff/5 或 √(ΔP) 公式。"""
    table = [(0, 0.0), (1, 1.0), (2, 1.5), (3, 2.0),
             (5, 3.0), (10, 3.5), (15, 4.0)]
    x = abs(abs_pressure_diff)
    if x >= 15:
        return 4.0
    for i in range(len(table) - 1):
        x0, y0 = table[i]
        x1, y1 = table[i + 1]
        if x0 <= x <= x1:
            return y0 + (y1 - y0) * (x - x0) / (x1 - x0)
    return 0.0

class CalcStep(ABC):
    @abstractmethod
    def get_inputs(self) -> list[str]: ...
    @abstractmethod
    def get_outputs(self) -> list[str]: ...
    @abstractmethod
    def execute(self, ctx: CalcContext) -> dict: ...
```

## 3. 静态冷量计算伪代码（F4-001）

```python
def static_calc_pipeline(room_id, water_temp_config_id=None):
    # Room 单表：负荷/风量/温湿度/新风/末端水温档参数全部直接读 room（PRD 附录A）
    ctx = CalcContext.load(room_id)

    # Step 1 焓值（室内默认 26°C / 55%；室外用国标夏季湿球）
    indoor_t = ctx.room.indoor_calc_temp or 26.0
    indoor_h = ctx.room.indoor_calc_humidity or 55.0     # 默认 55%，不是 60%
    h_out = calculate_enthalpy(ctx.city.temp_dry_ac_summer, ctx.city.temp_wet_ac_summer,
                               ctx.city.pressure_summer, 'wb')
    h_in  = calculate_enthalpy(indoor_t, indoor_h, ctx.city.pressure_summer, 'rh')
    ctx.set('RoomCalcResult.outdoor_enthalpy_calc', h_out)
    ctx.set('RoomCalcResult.indoor_enthalpy_calc', h_in)
    ctx.set('RoomCalcResult.enthalpy_diff', h_out - h_in)

    # Step 2 末端负荷（5 项；电动/电热为 kW 不乘 1000，空取 0）
    r = ctx.room
    civil    = r.civil_load_index * r.area / 1000
    lighting = r.lighting_load_index * r.area / 1000
    personnel = (r.personnel_load_index or 0) * (r.personnel_count or 0) / 1000
    electric  = (r.electric_equipment_power or 0) * (r.electric_equipment_coefficient or 0)
    heated   = (r.heated_equipment_power_with_exhaust or 0) * (r.heated_equipment_coefficient_with_exhaust or 0) \
             + (r.heated_equipment_power_without_exhaust or 0) * (r.heated_equipment_coefficient_without_exhaust or 0)
    terminal = civil + lighting + personnel + electric + heated
    ctx.set('RoomCalcResult.terminal_load', terminal)

    # Step 3 风量：8 点查表 + 简单加法（不加 MAX，不加卫生/净化风量）
    total_exhaust = sum(x or 0 for x in [r.heat_exhaust_volume, r.acid_exhaust_volume,
                       r.alkali_exhaust_volume, r.organic_exhaust_volume, r.dust_exhaust_volume])
    k = lookup_infiltration_coeff(r.pressure_diff or 0)     # PRD §8.3 八点插值
    infiltration = ctx.room.volume * k
    fresh = total_exhaust + infiltration                    # A4 简单加法，无 MAX/卫生/净化
    ctx.set('RoomCalcResult.total_exhaust_volume', total_exhaust)
    ctx.set('RoomCalcResult.infiltration_air_volume', infiltration)
    ctx.set('RoomCalcResult.fresh_air_volume', fresh)

    # Step 4 新风冷负荷（唯一公式：定义 A；ρ=1.2）
    fresh_load = RHO * fresh * (h_out - h_in) / SEC_PER_HOUR   # 可为负
    ctx.set('RoomLoadByWaterTemp.fresh_air_load', fresh_load)

    # Step 5 房间总负荷（<0 截断为 0，与动态统一；极值 Q_min 取 >0 最小）
    total = fresh_load + terminal
    if total < 0:
        total = 0.0
    ctx.set('RoomLoadByWaterTemp.total_load', total)

    # 按水温分类（ADR-0001）：末端归 Room.terminal_water_temp_config_id（默认中温）；
    # 新风按 fresh_air 配置归档（单水温全归该档；两阶段按比例，h_intermediate 预留）
    classify_by_water_temp(ctx, terminal, fresh_load, total)

    # Step 6 多级聚合 Room→Floor→Building(+ExtraLoad)→Project(+ExtraLoad) → LoadSummary
    aggregate_to_load_summary(ctx)
    return ctx.persist()
```

```python
def simulate_weather_driven(project_id, years, room_ids=None):
    batch = create_calc_batch(project_id, 'weather_driven', years)
    weather = load_weather_series(project_id, years)          # WeatherRecord 序列
    # Q_terminal 从静态 RoomCalcResult.terminal_load 取（桥梁，计算一次静态动态共用）
    rooms = load_rooms_static(project_id, room_ids)           # terminal_load, fresh_air_volume, h_in
    T = len(weather)
    buf = []
    for t in range(T):
        h_out = vectorized_enthalpy(weather[t])              # F6-002
        for r in rooms:
            q_fresh = RHO * r.fresh_air_volume * (h_out - r.h_indoor) / SEC_PER_HOUR  # F6-004
            q_total = r.terminal_load + q_fresh                                       # F6-005
            if q_total < 0: q_total = 0.0                                                # F6-006
            buf.append((batch.id, r.id, None, None, 'room', r.terminal_water_temp_config_id,
                        'weather_driven', weather[t].datetime,
                        weather[t].temp_dry, h_out, q_fresh, r.terminal_load, q_total))
        if len(buf) >= 2000:
            flush_dynamic_load_hourly(buf); buf.clear()
    flush_dynamic_load_hourly(buf)
    aggregate_dynamic(batch.id)                              # 见 §5
    return batch.id
```

```python
def simulate_coefficient(project_id, years, room_ids=None):
    batch = create_calc_batch(project_id, 'coefficient', years)
    weather = load_weather_series(project_id, years)
    rooms = load_rooms_static(project_id, room_ids)
    # 查 RatioCoefficient 表（PRD §14.8 统一映射表），取负荷率=100% 列，按室外温度查表
    coeff = load_ratio_coefficient(project_id)               # ADR-0003
    buf = []
    for t in range(len(weather)):
        k = coeff.lookup(outdoor_temp=weather[t].temp_dry, rate=1.0)   # F6-008/009
        for r in rooms:
            q = r.q_static * k                                # F6-010：Q_dynamic(t)=Q_static×k(t)
            if q < 0: q = 0.0
            buf.append((batch.id, r.id, None, None, 'room', r.terminal_water_temp_config_id,
                        'coefficient', weather[t].datetime, weather[t].temp_dry,
                        None, None, r.q_static, q))
        if len(buf) >= 2000:
            flush_dynamic_load_hourly(buf); buf.clear()
    flush_dynamic_load_hourly(buf)
    return batch.id
```

```python
def aggregate_dynamic(batch_id):
    # 房间级按水温分类（对应 RoomLoadByWaterTemp 语义）
    INSERT INTO DynamicLoadHourly (... scope='room' ...)
    SELECT project_id, calc_batch_id, room_id, 'room', water_temp_config_id,
           simulation_type, ts, SUM(fresh_air_load), SUM(terminal_load), SUM(total_load)
    FROM DynamicLoadHourly
    WHERE calc_batch_id=%s AND scope='room'
    GROUP BY room_id, water_temp_config_id, ts;

    # 建筑级（楼层聚合 + extra_load）
    INSERT ... scope='building'
    SELECT ..., building_id, ts, SUM(total_load) + extra_load(building_id)
    FROM DynamicLoadHourly JOIN Room ON ... JOIN ExtraLoad ON ...
    WHERE scope='room' GROUP BY building_id, ts;

    # 项目级（所有建筑 + extra_load(building_id 空)）
    INSERT ... scope='project'
    SELECT ..., NULL, ts, SUM(total_load) + extra_load(NULL)
    GROUP BY ts;
```
