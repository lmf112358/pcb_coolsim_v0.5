## 1. 概述

前端以 React 函数组件 + React Query 组织数据获取与缓存。模板给出「房间参数表单」组件：props 类型、React Query 查询/变更 Hook、加载态、错误态、提交。该表单数据经 API 落库后，驱动 F3-001（2D 底图编辑器）与 F7-001（逐时曲线）等可视化组件。

## 3. React Query Hooks

```typescript
// hooks/useRoom.ts
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { api } from "../lib/api";

export const useRoom = (roomId: number) =>
  useQuery({
    queryKey: ["room", roomId],
    queryFn: async () => {
      const { data } = await api.get(`/api/v1/rooms/${roomId}/`);
      return data.data as Room;
    },
  });

export const useSaveRoom = () => {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (payload: Partial<Room>) =>
      api.put(`/api/v1/rooms/${payload.id}/`, payload),
    onSuccess: (_res, vars) => {
      qc.invalidateQueries({ queryKey: ["room", vars.id] });
    },
  });
};

export const useCalculateRoom = () =>
  useMutation({
    mutationFn: (vars: { roomId: number; seasonId: number }) =>
      api.post(`/api/v1/rooms/${vars.roomId}/calculate-static/`, {
        season_id: vars.seasonId,
      }),
  });
```

## 4. 组件模板（加载 / 错误 / 提交）

```tsx
// components/RoomParameterForm.tsx
import { useState } from "react";
import { useRoom, useSaveRoom, useCalculateRoom } from "../hooks/useRoom";
import type { RoomFormProps, CalcSummary } from "../types/room";

export function RoomParameterForm({ roomId, seasonId, onCalculated }: RoomFormProps) {
  const { data: room, isLoading, isError, error } = useRoom(roomId);
  const save = useSaveRoom();
  const calc = useCalculateRoom();
  const [form, setForm] = useState<Partial<Room>>({});

  if (isLoading) return <div>加载中...</div>;
  if (isError) return <div>加载失败：{(error as Error).message}</div>;

  const onChange = (key: keyof Room, value: unknown) =>
    setForm((f) => ({ ...f, [key]: value }));

  const onSubmit = async () => {
    await save.mutateAsync({ id: roomId, ...form });
    const res = await calc.mutateAsync({ roomId, seasonId });
    onCalculated?.(res.data.data as CalcSummary);   // 驱动 F3/F7 可视化
  };

  return (
    <form onSubmit={(e) => { e.preventDefault(); onSubmit(); }}>
      <label>房间名称
        <input value={form.room_name ?? room!.room_name}
               onChange={(e) => onChange("room_name", e.target.value)} required />
      </label>
      <label>面积(m2)
        <input type="number" value={form.area ?? room!.area}
               onChange={(e) => onChange("area", Number(e.target.value))} required />
      </label>
      <label>高度(m)
        <input type="number" value={form.height ?? room!.height}
               onChange={(e) => onChange("height", Number(e.target.value))} required />
      </label>
      <p>体积(自动): {room!.volume} m3</p>
      {save.isError && <p className="error">保存失败</p>}
      <button type="submit" disabled={save.isLoading || calc.isLoading}>
        {save.isLoading ? "保存中..." : "保存并计算"}
      </button>
    </form>
  );
}
```
